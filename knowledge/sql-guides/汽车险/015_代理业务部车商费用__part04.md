SELECT distinct a.*
       , b.AGENT_CODE 代理点代码
       , b.agent_name 代理点名称 -----要匹配
       , b.department_group_name 部门组    
       , b.department_name 部门
       , c.rzbf 入账保费
       , b.tb_name 投保人
       , nvl(b.bb_name,b.tb_name) 被保险人
       , b.product_code 险种代码
       , b.product_name 险种名称 -----要匹配
       , b.inception_date 起保日期
       , b.planned_end_date 终止日期
       , b.operator_code 经办人代码
       , b.operator_name 经办人名称 -----要匹配
FROM tmp_mky_001_2 a
join etl_nonauto_policy_t b on a.policyno = b.policy_no 
left join(select policy_no,endorsement_no,sum(rzbf_notax) as rzbf
          from etl_payment_due_t
          GROUP BY policy_no,endorsement_no )c on a.policyno=c.policy_no AND a.endorseno=c.endorsement_no
where exists(select 1 from ids_wuh.NONAUTO_AGREEMENT_T c where a.policyno=c.policy_no);起保时间，已付佣金SELECT 
    a.department_group_code AS 部门组代码,
    a.department_group_name AS 部门组名称,
    COALESCE(a.department_code, '') AS 部门代码,
    COALESCE(a.department_name, '') AS 部门名称,
    a.product_code AS 险种代码,
    a.product_name AS 险种名称,
    a.agent_code AS 代理点代码,
    COALESCE(a.dldmc, '') AS 代理点名称,
    a.selling_channel_type AS 展业方式代码,
    a.selling_channel_name AS 展业方式名称,
    COALESCE(a.selling_shop_code, '') AS 车商代码,
    COALESCE(a.selling_shop_name, '') AS 车商名称,
    COUNT(a.policy_no) AS 保单数,
    SUM(COALESCE(a.premium_amount, 0) - COALESCE(a.tax_amount, 0)) AS 保费,
    SUM((COALESCE(a.premium_amount, 0) - COALESCE(a.tax_amount, 0)) * COALESCE(c.COMMISSIONRATE, 0)) AS 佣金,
    SUM(COALESCE(e.已付手续费, 0)) AS 已付佣金
