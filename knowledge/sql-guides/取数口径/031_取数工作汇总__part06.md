 理赔摊回赔款SELECT 
    -- policy_no  
    --, INDEMNITY_NO 
    CLIENT_NAME,
    (DEALED_AMOUNT - PENDING_AMOUNT) AS amount_owe,
    (
        SELECT NAME 
        FROM party_t b 
        WHERE b.policy_no = a.policy_no 
        AND b.role = '1'
    ) AS tb_name,
    OPERATION_DATE,
    (
        SELECT MAX(CLAIM_COST_CODE) 
        FROM NONAUTO_CLAIM_COST_T b 
        WHERE b.INDEMNITY_NO = a.INDEMNITY_NO
    ) AS amount_type_code,              -- 赔款：00、01000、01001、01002
    (
        SELECT f_basic_tjbmzmc(department_group_code, department_code) 
        FROM AUTO_CF_INDEMNITY_T b 
        WHERE b.indemnity_no = a.indemnity_no
    ) AS department_group_name1,
    (
        SELECT f_basic_tjbmzmc(department_group_code, department_code) 
        FROM NONAUTO_CF_INDEMNITY_T b 
        WHERE b.indemnity_no = a.indemnity_no
    ) AS department_group_name2
FROM PAYMENT_GENERATION_T a
WHERE 
    pending_cause = '1' 
    AND pending_type IN ('386', '396')    -- 待付款表 pending_cause='1' 表示赔款
    AND indemnity_no NOT IN (
        SELECT indemnity_no 
        FROM fr_fgs_cw_gbth_delete_list
    )  -- 剔除18条重复赔款等bug数据
    AND (DEALED_AMOUNT - PENDING_AMOUNT) <> 0
    AND client_name LIKE '%国任财产保险股份有限公司武汉中心支公司%'
