SELECT department_group_code
       , department_group_name
       , department_code
       , department_name
       , xz AS product_code
       , SUM(bf) AS wdybf
FROM etl_policy_request_issue_t WHERE nf = SUBSTR(etl_batch_date,0,4)-1 AND yf <= SUBSTR(etl_batch_date,5,2)-1
--AND workdate_premium <= to_char(last_day(add_months(to_date(etl_batch_date,'yyyyMMdd'),-13)),'yyyyMMdd')    
GROUP BY department_group_code,department_group_name,department_code,department_name,xz
)
SELECT a.department_group_code
       , a.department_group_name
       , a.department_code
       , a.department_name
       , a.product_code
       , (a.wdybf-nvl(b.wdybf,0)) wdybf
FROM case1 a LEFT JOIN case2 b
ON a.department_group_code = b.department_group_code
AND a.department_code = b.department_code
AND A.DEpartment_NAME = B.department_NAME
AND a.product_code = b.product_code;
END IF;
