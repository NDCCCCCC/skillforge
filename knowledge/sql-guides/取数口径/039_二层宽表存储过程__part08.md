 insert into v_nonauto_agreement_t4
 WITH RYX_AGREEMENT_T AS (
    SELECT 
        policy_no,
        COUNT(*) AS msb_1
    FROM 
        ids_wuh.ryx_agreement_t
    WHERE 
        partner_name IN ('A2系统-码上保', 'A2系统')
    GROUP BY 
        policy_no
),
CIBS_AGREEMENT_T AS (
    SELECT 
        policy_no,
        COUNT(*) AS msb_2
    FROM 
        ids_wuh.cibs_agreement_t
    WHERE 
        partner_name IN ('A2系统-码上保', 'A2系统')
    GROUP BY 
        policy_no
),
AUTO_AGREEMENT_EXTEND_T AS (
    SELECT 
        policy_no,
        COUNT(*) AS msb_3
    FROM 
        ids_wuh.auto_agreement_extend_t
    WHERE 
        tppartner_code = 'MSHB'
    GROUP BY 
        policy_no
),
RYX_TBL_PLCINFO_ADD_T AS (
    SELECT 
        TBDH,
        MIN(TASK_ID) AS TASK_ID_ryx,
        MAX(CUSTOMERDEPT) AS CUSTOMERDEPT2
    FROM 
        ids_wuh.RYX_TBL_PLCINFO_ADD_T
    GROUP BY 
        TBDH
),
CIBS_P_POLICY_EXTEND_T AS (
    SELECT 
        applyno,
        MIN(TASKNO) AS TASK_ID_cibs,
        MAX(CUSTOMERDEPT) AS CUSTOMERDEPT1
    FROM 
        ids_wuh.CIBS_P_POLICY_EXTEND_T
    GROUP BY 
        applyno
),
TYPECOMP_T AS (
    SELECT 
        yxzh,
        MAX(cwxzh) AS cwxzh
    FROM 
        ids_wuh.typecomp_T
    GROUP BY 
        yxzh
)
SELECT 
    a.*,
    F_BASIC_ISMSB(f1.msb_1, f2.msb_2, f3.msb_3) AS is_msb,
    F_BASIC_FC_CUSTOMERTYPE(a.policy_application_no, a.product_code, d2.CUSTOMERDEPT1, d1.CUSTOMERDEPT2) AS kq,
    F_BASIC_FC_CUSTOMERTYPE_NEW(g.CUSTOMERDEPT) AS kq_new,
    f_basic_xzhdl(h.cwxzh) AS xzhdl, -- 明细险种大类
    f_basic_xzhdlmc(h.cwxzh) AS xzhdlmc -- 明细险种大类名称
FROM 
    v_nonauto_agreement_t3 a
LEFT JOIN 
    RYX_TBL_PLCINFO_ADD_T d1 ON a.policy_application_no = d1.TBDH
LEFT JOIN 
    CIBS_P_POLICY_EXTEND_T d2 ON a.policy_application_no = d2.applyno
LEFT JOIN 
    RYX_AGREEMENT_T f1 ON a.policy_no = f1.policy_no
LEFT JOIN 
    CIBS_AGREEMENT_T f2 ON a.policy_no = f2.policy_no
LEFT JOIN 
    AUTO_AGREEMENT_EXTEND_T f3 ON a.policy_no = f3.policy_no
LEFT JOIN 
    ids_wuh.nonauto_agreement_t g ON a.policy_no = g.policy_no
LEFT JOIN 
    TYPECOMP_T h ON a.product_code = h.yxzh;
    v_nonauto_agreement_t4_num := sql%rowcount;
  if v_nonauto_agreement_t_num <> v_nonauto_agreement_t4_num then
    RETURN;
  end if;
