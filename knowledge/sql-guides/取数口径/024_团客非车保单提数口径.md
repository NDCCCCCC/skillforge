# 提数范围：起保年度在23年及以后的团客非车保单，剔除健康险、学平险。          提数字段：保单号、 保单起期、 保单止期、 太保分、 险种大类、 明细险种名称、 入账保费、 赔款金额（含已决和未决）。

>[!NOTE] 原笔记标题即为说明，正文以 SQL 为主。


**涉及表/视图**: `ids_wuh.nonAUTO_CF_INDEMNITY_T`, `wuh_dw.etl_nonauto_policy_t`, `ids_wuh.AUTO_CF_INDEMNITY_T`, `wuh_dw.etl_policy`, `wuh_dw.etl_payment_due_t`, `ids_wuh.RYX_TBL_PLCINFO_ADD_T`, `ids_wuh.CIBS_P_POLICY_EXTEND_T`, `tmp_cc_001`

## SQL / 正文

```sql
drop table tmp_cc_001;create table tmp_cc_001 asselect s.*,t.department_group_name 部门组名称,t.department_name 部门名称,t.section_office_name 科室名称,t.xzhdlmc 险种分类,t.kq_new 客群from (--已决清单select b.claim_folder_no 立案号,b.policy_no 保单号,b.posted_date 赔付日期,	   b.coinsurance_rate 共保比例,b.department_group_code 部门组代码,		b.department_code 部门代码,b.section_office_code 科室代码, sum(b.insured_object_indemnity+b.investigation_fee           +b.investigation_agent_fee+b.inside_investigation_agent_fee           +b.claim_audit_fee+b.salvage_fee+b.other_fee           -b.salvage_recovery) 赔付金额from ids_wuh.nonAUTO_CF_INDEMNITY_T b --非车险group by b.claim_folder_no,b.policy_no,b.posted_date,b.coinsurance_rate,b.department_group_code,b.department_code,b.section_office_code) sleft join wuh_dw.etl_nonauto_policy_t t on s.保单号=t.policy_nounion allselect s.*,t.department_group_name 部门组名称,t.department_name 部门名称,t.section_office_name 科室名称,t.product_name 险种分类,null 客群from (select b.claim_folder_no 立案号,b.policy_no 保单号,b.posted_date 赔付日期,	   0 共保比例,b.department_group_code 部门组代码,		b.department_code 部门代码,b.section_office_code 科室代码, sum(b.insured_object_indemnity+b.investigation_fee           +b.investigation_agent_fee+b.inside_investigation_agent_fee           +b.claim_audit_fee+b.salvage_fee+b.other_fee           -b.salvage_recovery) 赔付金额from ids_wuh.AUTO_CF_INDEMNITY_T b --非车险group by b.claim_folder_no,b.policy_no,b.posted_date,b.department_group_code,b.department_code,b.section_office_code) sleft join wuh_dw.etl_policy t on s.保单号=t.policy_no;drop table tmp_tkyyfwb_cl_001;create table tmp_tkyyfwb_cl_001 as (select a.policy_no 保单号,t.INCEPTION_DATE 保单起期,t.PLANNED_END_DATE 保单止期,r.tbf 太保分,c.score 太保分2,nvl(r.tbf,c.score) 太保分二合一,a.xzdl 险种大类,a.product_name 险种名称,t.xzhdlmc 险种分类,a.rzbf_with_tax 含税入账保费,a.rzbf_notax 不含税入账保费,m.yj 已决,i.gsje 未决from wuh_dw.etl_payment_due_t aleft join wuh_dw.etl_nonauto_policy_t t on a.policy_no=t.policy_noleft join ids_wuh.RYX_TBL_PLCINFO_ADD_T r on t.POLICY_APPLICATION_NO=r.tbdhleft join ids_wuh.CIBS_P_POLICY_EXTEND_T c on t.POLICY_APPLICATION_NO = c.applyno and applyendorseno = '无'left join (select 保单号, sum(赔付金额) yj from tmp_cc_001 group by 保单号) m     on a.policy_no = m.保单号left join (select nf,qsh,sum(gsje) gsje,polno from ids_wuh.INDISPUTE_T_1214 where nf='2025' and qsh='11'             group by polno,nf,qsh) i           on a.policy_no = i.polnoleft join wuh_dw.etl_nonauto_policy_t p on a.policy_no = p.policy_nowhere a.dd>=20230101and a.dd<=202501231and a.rzbf_notax<>0and a.xzdl = '非车险'and (a.product_name not like '%学生%' and a.product_name not like '%监护人%')and t.xzhdlmc != '十、短期健康保险'and p.class_name = '团客');select * from tmp_tkyyfwb_cl_001
```
