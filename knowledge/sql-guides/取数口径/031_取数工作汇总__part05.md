drop table tmp_ycb_001;
create table tmp_ycb_001 as
with v_case_1 as(
SELECT
a.policy_no 保单号,
a."endorsement_no" 批单号,
a.department_grouP_code 部门组代码,
a.department_group_name 部门组,
a.department_code 部门代码,
a.TJBMMC AS 部门,
sum(a.rzbf_notax) 入账保费,
max(to_date(a.dd,'yyyy-mm-dd')) 入账时间
FROM etl_payment_due_t a
where a.dd>=20240901
and a.dd<=20241031
group by a.policy_no,
a."endorsement_no",
a.department_grouP_code,
a.department_group_name,
a.department_code,
a.TJBMMC
)
select a.*
,t.kq 客群,t.operator_name 经办人,t.operator_code 经办人工号,t.product_code 险种代码,t.product_name 险种名称
,t.xzhdlmc 险种大类,t.insured_amount 保单保额,t.issue_date 保单签单日期,t.inception_date 保单起保日期,t.planned_end_date 保单终止日期
from v_case_1 a,etl_nonauto_policy_t t
where a.保单号=t.policy_no
and t.product_code in ('23XK9800','23C49800')
;

drop table tmp_ycb_002;
create table tmp_ycb_002 as
select a.*
       ,(select sum(nvl(extractamount, 0)) from "ids_wuh".cmextractcommission_t b 
                           where b.pOLICYNO= a.保单号 
                             and nvl(b.endoresno,'无')=a.批单号 
                             and b.commissiontype in ('0','1')) 计提手续费
       ,(select max(extractdate) from "ids_wuh".cmextractcommission_t b 
                           where b.pOLICYNO= a.保单号 
                             and nvl(b.endoresno,'无')=a.批单号 
                             and b.commissiontype in ('0','1')) 手续费计提时间,
       /*,(select sum(nvl(extractamount, 0)) from "ids_wuh".cmextractcommission_t b 
                           where b.pOLICYNO= a.保单号 
                             and nvl(b.endoresno,'无')=a.批单号 
                             and b.commissiontype in ('2')) 计提绩效
       ,(select max(extractdate) from "ids_wuh".cmextractcommission_t b 
                           where b.pOLICYNO= a.保单号 
                             and nvl(b.endoresno,'无')=a.批单号 
                             and b.commissiontype in ('2')) 绩效计提时间
       ,(select sum(nvl(extractamount, 0)) from "ids_wuh".cmextractcommission_t b 
                           where b.pOLICYNO= a.保单号 
                             and nvl(b.endoresno,'无')=a.批单号 
                             and b.commissiontype in ('4')) 计提业务费
       ,(select max(extractdate) from "ids_wuh".cmextractcommission_t b 
                           where b.pOLICYNO= a.保单号 
                             and nvl(b.endoresno,'无')=a.批单号 
                             and b.commissiontype in ('4')) 业务费计提时间
       --, (SELECT SUM(SIGNPREMIUM*DOCUMENTARYCOMMISSIONRATE) FROM "ids_wuh".CSCMFAUTPOLICYFEE b WHERE  b.pOLICYNO  = a.保单号 and nvl(b.ENDORSENO,'无')=a.批单号 AND status = '20') 跟单手续费   */
       , (SELECT SUM(SIGNPREMIUM*DOCUMENTARYBUSINESSRATE) FROM "ids_wuh".CSCMFAUTPOLICYFEE b WHERE  b.pOLICYNO  = a.保单号 and nvl(b.ENDORSENO,'无')=a.批单号 AND status = '20') 跟单业务费
       , (SELECT SUM(SIGNPREMIUM*DOCUMENTARYBONUSRATE) FROM "ids_wuh".CSCMFAUTPOLICYFEE b WHERE  b.pOLICYNO  = a.保单号 and nvl(b.ENDORSENO,'无')=a.批单号 AND status = '20') 跟单绩效 
from tmp_ycb_001 a
