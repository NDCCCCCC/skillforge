insert into FR_FGS_KMH_2025_sign_tmp_003 
SELECT a.*
       , DECODE(department_group_name,'神农架支公司','宜昌中心支公司',department_group_name) bmz
       , (CASE when xzdl <> '非车险' THEN ''
        else
         case when product_code='13093300'
            then '个信保分部' else
               case when a.label_name<>'无'
                 then '渠道管理部'  else
                   case when a.kq_new='政保健康'
                     then '健康险事业部'
                       when a.kq_new='政保客户' then
                         case when a.xzhdlmc='十、短期健康保险'
                           then '健康险事业部' else '政保业务部' end
                             else case when product_name in (select product_name from FGS_FR_INFOWIN_product_list_jkx)
                               then '健康险事业部' else '企业客户部' end
                                 end
                                 end
                                 end
                                 end)lines
FROM FR_FGS_KMH_2025_sign_tmp_002 a;
