select t.nf 年,t.qsh 月,t.caseno 报案号,t.indemno 赔案号,t.bzhdm 币种代码,
       t.gsje 年末估损金额,t.xzhdm 险种代码,t.polno 保单号,t.qbrq 起保日期,
       s.ISSUE_DATE 签单日期,coinsurance_indicator_name 共保标志,COINSURANCE_PERCENT 共保比例,
       s.SELLING_CHANNEL_NAME 展业方式,

       t.zlje 自留金额,s.xzdl 险种大类,null 使用性质,null 车辆类型
from INDISPUTE_T t,etl_nonauto_policy_t@ora_247 s

and t.qsh='08'
and t.polno=s.policy_no
and s.xzdl='非车险'
union all
select t.nf 年,t.qsh 月,t.caseno 报案号,t.indemno 赔案号,t.bzhdm 币种代码,
       t.gsje 年末估损金额,t.xzhdm 险种代码,t.polno 保单号,t.qbrq 起保日期,
       s.ISSUE_DATE 签单日期,null 共保标志,null 共保比例,
       s.SELLING_CHANNEL_NAME 展业方式,
       t.zhzhrq 终止日期,t.chxrq 出险日期,t.larq 立案日期,
       t.bbrmc 被保险人名称,t.gsry 估损人员,t.gsrymc 估损人员姓名,t.ywry 业务人员,t.ywrymc 业务人员名称,t.bmdm 部门代码,
       t.bmmc 部门名称,t.kshdm 科室代码,t.kshmc 科室名称,t.dlddm 代理点代码,t.dldmc 代理点名称,
       t.zlje 自留金额,'车险' 险种大类,usage_name 使用性质,vehicle_type_name 车辆类型
