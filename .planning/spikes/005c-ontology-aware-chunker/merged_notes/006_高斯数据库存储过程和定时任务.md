CREATE OR REPLACE FUNCTION wuh_dw.pro_qdrb0_bt01_gfc_and_xny()
 RETURNS void
 LANGUAGE plpgsql
 NOT FENCED NOT SHIPPABLE
AS $$
DECLARE
    v_count INT;
BEGIN
/*
 * BT01实时签单报表：个非车和新能源实时签单保费
*/

    -- 新增逻辑：检查目标表是否存在，不存在则创建
    SELECT COUNT(*)
    INTO v_count
    FROM information_schema.tables
    WHERE table_schema = 'wuh_dw'
      AND table_name = 'pro_qdrbt0_bt01_gfc_and_xny';

    IF v_count = 0 THEN
        EXECUTE IMMEDIATE '
            CREATE TABLE wuh_dw.pro_qdrbt0_bt01_gfc_and_xny (
                department_group_code VARCHAR(50),
                department_group_name VARCHAR(100),
                department_code VARCHAR(50),
                department_name VARCHAR(100),
                xz VARCHAR(20),
                fueltype VARCHAR(50),
                customerdept VARCHAR(50),
                bf DECIMAL(15, 2)
            )
        ';
    END IF;

--更新BT01实时签单
EXECUTE IMMEDIATE 'TRUNCATE TABLE PRO_QDRBT0_BT01_GFC_AND_XNY';
INSERT INTO PRO_QDRBT0_BT01_GFC_AND_XNY
WITH case_data AS (
SELECT 
        a.policyno AS policy_no,
        a.departmentGroupCode,
        a.departmentCode,
        DECODE(a.autoFlg,'3','车险','7','农险','非车险')as xz,
        a.fueltype,
        a.customerdept,
        SUM(
            (CASE 
                WHEN a.autoflg = '3' AND NVL(a.endorsementno, '无') = '无' THEN NVL(b.totalPremiumChange, 0) - NVL(b.taxAmount, 0)
                WHEN a.autoflg = '3' AND NVL(a.endorsementno, '无') <> '无' THEN NVL(b.totalPremiumChange, 0) - NVL(b.PREMIUMCHANGETAX, 0)
                WHEN a.autoflg <> '3' AND NVL(a.endorsementno, '无') = '无' THEN NVL(b.BAKTOTALPREMIUM, 0) - NVL(b.TAXAMOUNT, 0)
                WHEN a.autoflg <> '3' AND NVL(a.endorsementno, '无') <> '无' THEN NVL(b.TOTALPREMIUMCHANGE, 0) - NVL(b.PREMIUMCHANGETAX, 0)
                ELSE 0 
            END) * cc.rate / 100
        ) AS bf
    FROM 
        ids_wuh.msg_dwa_policy_summarys a
    JOIN 
        ids_wuh.msg_dwa_policy_summarys_tot_prem b 
        ON a.data_id = b.data_parent_id
    JOIN 
        ids_wuh.exrate_month_t cc 
        ON b.origcurrency = cc.bzh 
        AND cc.theyear || LPAD(cc.themonth, 2, '0') = TO_CHAR(SYSDATE, 'yyyymm')
    WHERE 
        a.signatureDate LIKE TO_CHAR(SYSDATE, 'yyyy-mm-dd') || '%'
        and a.autoflg != '7'
    GROUP BY 
        a.policyno, a.departmentGroupCode, a.departmentCode, a.autoFlg, a.fueltype, a.customerdept
),
department_info AS (

    SELECT 
        a.departmentGroupCode,
        a.departmentCode as department_code,
        f_basic_tjbmzdm(a.departmentGroupCode, a.departmentCode) AS department_group_code,
        f_basic_tjbmzmc(a.departmentGroupCode, a.departmentCode) AS department_group_name,
        f_basic_tjbmmc(a.departmentGroupCode, a.departmentCode) AS department_name,
        a.xz,
        a,fueltype,
        a.customerdept,
        a.bf
    FROM 
        case_data a
)   
SELECT department_group_code,department_group_name,department_code,department_name,xz,fueltype,customerdept,SUM(bf) bf FROM department_info GROUP BY department_group_code,department_group_name,department_code,department_name,xz,fueltype,customerdept;

END$$

-- 假设当前计算日期为 '2025-01-09'，您可以通过变量或函数动态获取
WITH current_date_cte AS (
    SELECT DATE('${yearmonth}') AS calc_date
),
-- 将原始日期字段转换为标准日期，并提取年、月维度
date_converted AS (

    SELECT 
        *,
        -- 将数字格式的dd转换为日期，例如 20250106 -> 2025-01-06
        -- 使用CAST将dd转换为VARCHAR类型，然后使用TO_DATE转换为日期
        TO_DATE(CAST(dd AS VARCHAR), 'yyyyMMdd') AS sign_date,
        EXTRACT(YEAR FROM TO_DATE(CAST(dd AS VARCHAR), 'yyyyMMdd')) AS sign_year,
        EXTRACT(MONTH FROM TO_DATE(CAST(dd AS VARCHAR), 'yyyyMMdd')) AS sign_month,
        EXTRACT(DAY FROM TO_DATE(CAST(dd AS VARCHAR), 'yyyyMMdd')) AS sign_day
    FROM wuh_dw.FR_FGS_signature_005
    WHERE bf IS NOT NULL AND dd IS NOT NULL -- 确保关键字段不为空
      and fuel_type = '新能源车'
      -- 过滤掉运营服务部和个信保
      AND department_group_name NOT LIKE '%运营服务本部%'
      AND department_group_name NOT LIKE '%个信保%'
),
-- 创建分公司本部映射表（根据您的实际业务规则调整）
department_mapping AS (
     SELECT 
        department_group_name,
        -- 提取分公司本部名称：假设部门名称中包含"中心支公司"或"支��司"的为分公司本部
        CASE 
            WHEN department_group_name = '业务' THEN 
                '分公司本部' -- 提取中心支公司名称
            when department_group_name like '%支公司%' then
              REPLACE(REPLACE(department_group_name, '中心支公司', ''), '支公司', '')
            ELSE department_group_name -- 如果没有匹配，保持原样
        END AS branch_headquarters
    FROM date_converted
    GROUP BY department_group_name
),
ssbf AS (

select department_group_name,sum(bf) as bf from "pro_qdrbt0_bt01_gfc_and_xny" where fueltype in ('1','3','4') and xz = '车险' group by department_group_name
),
-- 计算当日保费，按分公司本部分组
daily_premium AS (
    SELECT 
        dm.branch_headquarters,
        SUM(d.bf)/10000 AS daily_premium
--    FROM date_converted d
    from ssbf d
    JOIN department_mapping dm ON d.department_group_name = dm.department_group_name
    GROUP BY dm.branch_headquarters
),
-- 计算当月保费 (截至计算日所在月份)，按分公司本部分组
monthly_premium_current AS (
    SELECT 
        dm.branch_headquarters,
        d.sign_year,
        d.sign_month,
        (SUM(d.bf)+ss.daily_premium)/10000 AS monthly_premium
    FROM date_converted d
    JOIN department_mapping dm ON d.department_group_name = dm.department_group_name
    left join daily_premium ss on ss.branch_headquarters=dm.branch_headquarters
    CROSS JOIN current_date_cte c
    WHERE d.sign_year = EXTRACT(YEAR FROM c.calc_date)
      AND d.sign_month = EXTRACT(MONTH FROM c.calc_date)
    GROUP BY dm.branch_headquarters, d.sign_year, d.sign_month, ss.daily_premium
),
-- 计算去年同月保费，用于同比，按分公司本部分组
monthly_premium_last_year AS (
    SELECT 
        dm.branch_headquarters,
        d.sign_year,
        d.sign_month,
        SUM(d.bf)/10000 AS monthly_premium_ly
    FROM date_converted d
    JOIN department_mapping dm ON d.department_group_name = dm.department_group_name
    CROSS JOIN current_date_cte c
    WHERE d.sign_year = EXTRACT(YEAR FROM c.calc_date) - 1
      AND d.sign_month = EXTRACT(MONTH FROM c.calc_date)
      and cast(d.sign_day as int) <= cast(EXTRACT(DAY FROM c.calc_date) as int)
    GROUP BY dm.branch_headquarters, d.sign_year, d.sign_month
),
-- 计算当年累计保费，按分公司本部分组
yearly_premium_current AS (

    SELECT 
        dm.branch_headquarters,
        d.sign_year,
        (SUM(d.bf)+ss.daily_premium)/10000 AS yearly_premium
    FROM date_converted d
    JOIN department_mapping dm ON d.department_group_name = dm.department_group_name

-- 最终结果整合，按分公司本部分组并排序
SELECT 
    ab.branch_headquarters AS department_group_name,
--    (SELECT calc_date FROM current_date_cte) AS statistic_date,
    COALESCE(dp.daily_premium, 0) AS daily_premium,
    COALESCE(mpc.monthly_premium, 0) AS monthly_premium,
    COALESCE(mpl.monthly_premium_ly, 0) AS monthly_premium_ly,
    -- 计算当月同比，处理除零错误
    CASE 
        WHEN COALESCE(mpl.monthly_premium_ly, 0) = 0 THEN NULL 
        ELSE ROUND((COALESCE(mpc.monthly_premium, 0) - COALESCE(mpl.monthly_premium_ly, 0)) / COALESCE(mpl.monthly_premium_ly, 0), 3)
    END AS monthly_premium_yoy_rate,
    COALESCE(ypc.yearly_premium, 0) AS yearly_premium,
    COALESCE(ypl.yearly_premium_ly, 0) AS yearly_premium_ly,
    -- 计算当年同比，处理除零错误
    CASE 
        WHEN COALESCE(ypl.yearly_premium_ly, 0) = 0 THEN NULL 
        ELSE ROUND((COALESCE(ypc.yearly_premium, 0) - COALESCE(ypl.yearly_premium_ly, 0)) / COALESCE(ypl.yearly_premium_ly, 0), 3)
    END AS yearly_premium_yoy_rate
FROM all_branches ab
LEFT JOIN daily_premium dp ON ab.branch_headquarters = dp.branch_headquarters
LEFT JOIN monthly_premium_current mpc ON ab.branch_headquarters = mpc.branch_headquarters
LEFT JOIN monthly_premium_last_year mpl ON ab.branch_headquarters = mpl.branch_headquarters
LEFT JOIN yearly_premium_current ypc ON ab.branch_headquarters = ypc.branch_headquarters
LEFT JOIN yearly_premium_last_year ypl ON ab.branch_headquarters = ypl.branch_headquarters
-- 按分公司本部名称排序
ORDER BY ab.branch_headquarters;
