# 逐单未到期准备金



**涉及表/视图**: `FR_FGS_ZDWDQ_PRODUCTCODE`, `op_uprcollection_all_t_adj`, `TMP_WANG_0011`, `TMP_WANG_0022`, `TMP_WANG_0033`, `TMP_WANG_0044`, `TMP_WANG_0055`

**输出字段示例**: 整体, 车险, 非车非农, 农险, 车意险险种大类, 整体, 车险, 非车非农

## 说明

## 说明



准备金（精算准备金）是保险公司为了履行保单责任及其相关支出所做的资金准备，或者说是对保险公司有效保险合同的保险责任金额，是保险公司最主要的负债准备金分为 未到期责任准备金 和 未决赔款准备金 两种维度（意义不一样）未到期对应保费责任类似会计上的权责发生制，过1个月，确认保单1/12的份额未决对应赔款责任未来的赔付风险分公司只下发了现成的未到期准备金数据。而后者是精算部建模、加权预测出来的。逐单未到期是指一单一单的算未到期准备金，但是数据口径包括法定未到期和报表未到期法定未到期 扣除 首日费用率 = 报表未到期（业务常用口径，也称2号）未到期计算方式根据表 OP_UPRCOLLECTION_ALL_T_ADJ（卡评估时点算全量）法定未到期：ADJUST_AFTER_BRANCHREIN_UPR+ ADJUST_LSBPI_UPR（调整后法定分公司再保后|未到期+保费不足）报表未到期：ADJUST_AFTER_BRANCHREIN_UPR2 + ADJUST_LSBPI_UPR2（调整后二号分公司再保后|未到期+保费不足）根据法定未到期得到报表未到期--逐单未到期是指一单一单的算未到期，包括法定未到期和报表未到期

--法定未到期 - 首日费用率 = 报表未到期

DROP TABLE TMP_WANG_0011 PURGE;

--罗瑜：法定未到期 分机构和险种 调整后法定分公司再保后未到期+调整后法定分公司再保后保费不足
INSERT INTO fr_fgs_fdwdq_result_jgxz  NOLOGGING 
SELECT * FROM (
SELECT 评估时点
       , 部门组名称
    , 部门名称
    , '整体' 险种大类
       , SUM(调整后法定分公司再保后未到期+调整后法定分公司再保后保费不足) "调整后法定分公司再保后(未到期+保费不足)" 
FROM TMP_WANG_0011 a
GROUP BY 评估时点,部门组名称,部门名称,险种大类
union ALL

SELECT 评估时点
       , 部门组名称
    , 部门名称
    , (CASE WHEN 险种大类 IS NULL THEN '险种为空' ELSE 险种大类 END)险种大类
       , SUM(调整后法定分公司再保后未到期+调整后法定分公司再保后保费不足) "调整后法定分公司再保后(未到期+保费不足)" 
FROM TMP_WANG_0011 a
GROUP BY 评估时点,部门组名称,部门名称,险种大类
)
PIVOT
(SUM("调整后法定分公司再保后(未到期+保费不足)") FOR 险种大类 IN ('整体' AS 整体,'车险' AS 车险,'非车非农' AS 非车非农,'农险' AS 农险,'车意险险种大类' AS 车意险险种大类));
COMMIT;
