drop table tmp_mky_001_1;
create table tmp_mky_001_1 as
select a.policyno 
       , NVL(a.endorseno, '无') endorseno 
       , sum(a.chinacommission) chinacommission
       , b.paydate paydate
       , b.balancingcode 
       , b.applyno 
       , b.status
 from ids_wuh.Cmpolicybanlaning_t a
 join ids_wuh.CMBALANCINGAPPLY_T b on a.balancingserialno = b.balancingserialno
 where a.COMMISSIONTYPE in('0','1') --and b.status ='1' 
 group by a.policyno, NVL(a.endorseno,'无'),b.balancingcode,b.applyno,b.paydate, b.status;
 


drop table tmp_mky_001_2;
create table tmp_mky_001_2 as
select a.policyno 
     , a.endorseno 
     , b.手续费率 手续费费率
     , b.已结算手续费 手续费金额
     , a.chinacommission 已付手续费
     , a.paydate 支付日期  
     , a.balancingcode 结算单号
     , a.applyno 收付号
     from tmp_mky_001_1 a
     left join (select policyno,nvl(endorseno, '无') endorseno,
                      max(decode(commissiontype, '0', commissionrate, 0)) as 手续费率,
         sum(decode(commissiontype, '0', balancedcommission, 0)) as 已结算手续费
         from ids_wuh.cmsummary_t
         where COMMISSIONSUMMARYNO > 0 and status <> -1
         group by policyno,nvl(endorseno, '无') ) b on a.policyno=b.policyno and a.endorseno=b.endorseno

SELECT 
    a.department_group_code AS 部门组代码,
    a.department_group_name AS 部门组名称,
    COALESCE(a.department_code, '') AS 部门代码,
    COALESCE(a.department_name, '') AS 部门名称,
    a.product_code AS 险种代码,
    a.product_name AS 险种名称,
    a.agent_code AS 代理点代码,
    COALESCE(a.dldmc, '') AS 代理点名称,
    a.selling_channel_type AS 展业方式代码,
    a.selling_channel_name AS 展业方式名称,
    COALESCE(a.selling_shop_code, '') AS 车商代码,
    COALESCE(a.selling_shop_name, '') AS 车商名称,
    COUNT(a.policy_no) AS 保单数,
    SUM(COALESCE(a.premium_amount, 0) - COALESCE(a.tax_amount, 0)) AS 保费,
    SUM((COALESCE(a.premium_amount, 0) - COALESCE(a.tax_amount, 0)) * COALESCE(c.COMMISSIONRATE, 0)) AS 佣金,
    SUM(COALESCE(e.已付手续费, 0)) AS 已付佣金
FROM wuh_dw.etl_policy a 
LEFT JOIN (

    SELECT policyno, MAX(COMMISSIONRATE) AS COMMISSIONRATE
    FROM ids_wuh.CSCMAUTPOLICYFEE
    GROUP BY policyno
) c ON a.policy_no = c.policyno
LEFT JOIN (select policyno,sum(已付手续费) 已付手续费 from tmp_mky_001_2 where 支付日期 is not null group by policyno) e ON a.policy_no = e.policyno
WHERE a.inception_date BETWEEN TO_DATE('2025-01-01 00:00:00', 'yyyy-mm-dd hh24:mi:ss')
    AND TO_DATE('2025-12-31 23:59:59', 'yyyy-mm-dd hh24:mi:ss')
GROUP BY 
    a.department_group_code,
    a.department_group_name,
    a.department_code,
    a.department_name,
    a.product_code,
    a.product_name,
    a.agent_code,
    a.dldmc,
    a.selling_channel_type,
    a.selling_channel_name,
    a.selling_shop_code,
    a.selling_shop_name

UNION ALL

SELECT 
    a.department_group_code AS 部门组代码,
    a.department_group_name AS 部门组名称,
    COALESCE(a.department_code, '') AS 部门代码,
    COALESCE(a.department_name, '') AS 部门名称,
    a.product_code AS 险种代码,
    a.product_name AS 险种名称,
    a.agent_code AS 代理点代码,
    COALESCE(a.agent_name, '') AS 代理点名称,
    a.selling_channel_type AS 展业方式代码,
    a.selling_channel_name AS 展业方式名称,
    '' AS 车商代码,
    '' AS 车商名称,
    COUNT(a.policy_no) AS 保单数,
    SUM(COALESCE(a.premium_amount, 0) - COALESCE(a.tax_amount, 0)) AS 保费,
    SUM((COALESCE(a.premium_amount, 0) - COALESCE(a.tax_amount, 0)) * COALESCE(c.COMMISSIONRATE, 0)) AS 佣金,
    SUM(COALESCE(e.已付手续费, 0)) AS 已付佣金
FROM wuh_dw.etl_nonauto_policy_t a 
LEFT JOIN (

    SELECT policyno, MAX(COMMISSIONRATE) AS COMMISSIONRATE
    FROM ids_wuh.CSCMFAUTPOLICYFEE
    GROUP BY policyno
) c ON a.policy_no = c.policyno
LEFT JOIN (select policyno,sum(已付手续费) 已付手续费 from tmp_mky_001_2 where 支付日期 is not null group by policyno) e ON a.policy_no = e.policyno
WHERE a.inception_date BETWEEN TO_DATE('2025-01-01 00:00:00', 'yyyy-mm-dd hh24:mi:ss')
    AND TO_DATE('2025-12-31 23:59:59', 'yyyy-mm-dd hh24:mi:ss')
GROUP BY
