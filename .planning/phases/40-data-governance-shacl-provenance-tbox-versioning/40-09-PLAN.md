        }
    }
    
    // Count: all new triples should have provenance
    newCount := 0
    legacyCount := 0
    for _, tr := range all {
        if tr.Provenance != nil {
            newCount++
        } else {
            legacyCount++
        }
    }
    if newCount == 0 {
        t.Errorf("expected some triples with provenance")
    }
}

func TestRollbackLatency(t *testing.T) {
    if testing.Short() {
        t.Skip("skipping latency test in short mode")
    }
    
    tmpDir := t.TempDir()
    ns := "http://skillforge.local/ontology#"
    kg := store.New()
    
    // Setup: write v1 TTL + v2 TTL + proposals
    v1TTL := `@prefix skf: <` + ns + `> .
skf:Thing a skf:Class .
`
    v2TTL := `@prefix skf: <` + ns + `> .
skf:Thing a skf:Class .
skf:NewClass a skf:Class .
`
    os.WriteFile(filepath.Join(tmpDir, "skillforge_tbox.v1.ttl"), []byte(v1TTL), 0o644)
    os.WriteFile(filepath.Join(tmpDir, "skillforge_tbox.v2.ttl"), []byte(v2TTL), 0o644)
    os.WriteFile(filepath.Join(tmpDir, "skillforge_tbox.ttl"), []byte(v2TTL), 0o644)
    os.WriteFile(filepath.Join(tmpDir, "tbox_version.txt"), []byte("2\n"), 0o644)
    os.WriteFile(filepath.Join(tmpDir, "proposals.json"), []byte(`{"next_id":1,"items":[]}`), 0o644)
    
    // Create manager
    m, err := NewManager(ns, tmpDir, nil, kg)
    if err != nil {
        t.Fatalf("new manager: %v", err)
    }
    
    // Load v2 TTL into KG
    kg.LoadFromTurtle(v2TTL)
    
    // Measure rollback
    start := time.Now()
    newVersion, violations, err := m.RollbackTBox(1, "test rollback")
    elapsed := time.Since(start)
    
    if err != nil {
        t.Fatalf("rollback: %v", err)
    }
    if newVersion != 3 {
        t.Errorf("expected version 3, got %d", newVersion)
    }
    if violations < 0 {
        t.Errorf("violations should be non-negative, got %d", violations)
    }
    
    // SC-4: rollback < 5s
    if elapsed > 5*time.Second {
        t.Errorf("rollback took %v, expected < 5s", elapsed)
    }
    
    t.Logf("rollback took %v", elapsed)
}
