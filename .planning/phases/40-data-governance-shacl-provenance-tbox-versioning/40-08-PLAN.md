```go
func (s *Deps) adminOntologyKGQueryHandler(ctx context.Context, c *app.RequestContext) {
    dep, ok := s.requireOntology(c)
    if !ok { return }

    skill := c.Query("skill")
    var matches []store.Triple
    if skill != "" {
        // Map skill name to allowedTiers via reads field
        allowedTiers := s.resolveSkillTiers(skill)  // new helper
        matches = dep.KG.QueryScoped(c.Query("s"), c.Query("p"), c.Query("o"), allowedTiers)
    } else {
        matches = dep.KG.Query(c.Query("s"), c.Query("p"), c.Query("o"))
    }
    if matches == nil {
        matches = []store.Triple{}
    }
    c.JSON(http.StatusOK, kgQueryResponse{Matches: matches})
}
```
