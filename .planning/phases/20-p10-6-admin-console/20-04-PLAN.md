### Task 3: SkillManagementList component

**Files:** `web/src/components/admin/SkillManagementList.tsx`

**Action:** Create the skill management list component:

1. `useEffect` fetching `GET /api/admin/skills` with `X-Admin-Token` header
2. Render a `<table>` with columns: Name | Version | Status | Last Eval | Last Change
3. Hot-reload status dot (green/red/amber Circle icon from lucide-react)
4. Last Eval Summary: hover cell that shows tooltip on mouse enter (200ms delay), dismisses on mouse leave
5. Tooltip shows: faithfulness score, context_precision score, eval timestamp
6. Last Change formatted as relative time

```typescript
function SkillManagementList() {
  const [skills, setSkills] = useState<SkillRow[]>([]);
  const [loading, setLoading] = useState(true);
  const [hoveredSkill, setHoveredSkill] = useState<string | null>(null);

  useEffect(() => {
    const token = /* read from env or store */;
    fetch('/api/admin/skills', {
      headers: { 'X-Admin-Token': token }
    })
      .then(r => r.json())
      .then(d => { setSkills(d.skills); setLoading(false); })
      .catch(() => setLoading(false));
  }, []);

  if (loading) return <LoadingSkeleton />;
  if (skills.length === 0) {
    return (
      <div className="text-center py-12">
        <h3 className="text-lg font-semibold">暂无注册 Skill</h3>
        <p className="text-sm text-text-muted mt-1">创建 skills/*.d/SKILL.md 注册新 Skill</p>
      </div>
    );
  }

  return (
    <table>
      <thead>
        <tr>
          <th>Name</th>
          <th>Version</th>
          <th>Status</th>
          <th>Last Eval</th>
          <th>Last Change</th>
        </tr>
      </thead>
      <tbody>
        {skills.map(skill => (
          <tr key={skill.name}>
            <td className="font-medium">{skill.name}</td>
            <td className="font-mono text-xs">{skill.version}</td>
            <td>
              <StatusDot status={skill.status} />
            </td>
            <td
              className="relative cursor-pointer text-xs text-text-muted hover:text-text"
              onMouseEnter={() => setTimeout(() => setHoveredSkill(skill.name), 200)}
              onMouseLeave={() => setHoveredSkill(null)}
            >
              {skill.last_eval_summary ? 'View details' : 'No eval'}
              {hoveredSkill === skill.name && skill.last_eval_summary && (
                <SkillEvalTooltip summary={skill.last_eval_summary} />
              )}
            </td>
            <td className="text-sm text-text-muted">
              {skill.last_change_at ? formatRelativeTime(skill.last_change_at) : '—'}
            </td>
          </tr>
        ))}
      </tbody>
    </table>
  );
}

function StatusDot({ status }: { status: string }) {
  const colors = {
    ok: 'text-green-400',
    failed: 'text-red-400',
    pending: 'text-amber-400',
  };
  return <span className={cn('inline-block w-2 h-2 rounded-full', colors[status] || 'bg-gray-400')} />;
}

function SkillEvalTooltip({ summary }: { summary: EvalSummary }) {
  return (
    <div className="absolute z-50 bottom-full left-1/2 -translate-x-1/2 mb-2 bg-surface border border-border rounded-md shadow-lg p-3 text-xs w-48">
      <div className="font-semibold mb-2">Eval Results</div>
      <div className="space-y-1">
        <div>faithfulness: <span className="font-mono">{summary.faithfulness}</span></div>
        <div>context_precision: <span className="font-mono">{summary.context_precision}</span></div>
        <div className="text-text-muted mt-2">{formatDate(summary.eval_ts)}</div>
      </div>
    </div>
  );
}
```

Per UI-SPEC: eval results are hover tooltips ONLY, NOT inline badges.

**Verify:** Tooltip appears on hover (mouse enter), disappears on leave, 200ms delay, no inline badge
**Done:** SkillManagementList.tsx exists with table, status dots, and hover tooltip for eval results
