```go
type CaseResult struct {
    ID                string  `json:"id"`
    Category          string  `json:"category"`        // "adversarial" | "refusal-required" | "multi-hop"
    Faithfulness      float64 `json:"faithfulness"`
    ContextPrecision  float64 `json:"context_precision"`
    AnswerRelevance   float64 `json:"answer_relevance"`
    ContextRecall     float64 `json:"context_recall"`
    Refused           bool    `json:"refused"`
}
type Report struct {
    Skill       string                        `json:"skill,omitempty"`
    Overall     map[string]float64            `json:"overall"`        // 4 keys: faithfulness, context_precision, answer_relevance, context_recall — NO refusal_accuracy
    PerCategory map[string]map[string]float64 `json:"per_category"`
    Cases       []CaseResult                  `json:"cases"`
}
```
