```typescript
import { useState, useEffect, useCallback } from 'react';
import type { NeighborResult } from '../types/knowledge';

interface UseGraphNeighborsOptions {
  depth?: number;
  enabled?: boolean;
}

interface UseGraphNeighborsResult {
  data: NeighborResult | null;
  loading: boolean;
  error: string | null;
  refetch: (chunkId: string, depth?: number) => void;
}

export function useGraphNeighbors(
  initialChunkId?: string,
  initialDepth?: number,
  options: UseGraphNeighborsOptions = {}
): UseGraphNeighborsResult {
  const [chunkId, setChunkId] = useState<string | undefined>(initialChunkId);
  const [depth, setDepth] = useState<number>(initialDepth ?? 2);
  const [data, setData] = useState<NeighborResult | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const fetchNeighbors = useCallback(async (cid: string, d: number) => {
    if (!cid) return;
    setLoading(true);
    setError(null);
    try {
      const url = `/api/knowledge/graph/neighbors?chunk_id=${encodeURIComponent(cid)}&depth=${d}`;
      const resp = await fetch(url, { headers: { Accept: 'application/json' } });
      if (!resp.ok) {
        throw new Error(`HTTP ${resp.status}`);
      }
      const json: NeighborResult = await resp.json();
      setData(json);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'fetch failed');
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    if (chunkId && options.enabled !== false) {
      fetchNeighbors(chunkId, depth);
    }
  }, [chunkId, depth, fetchNeighbors, options.enabled]);

  const refetch = useCallback((cid: string, d?: number) => {
    setChunkId(cid);
    if (d !== undefined) setDepth(d);
    void fetchNeighbors(cid, d ?? depth);
  }, [depth, fetchNeighbors]);

  return { data, loading, error, refetch };
}
```

The hook is a standard React data-fetching pattern: fetch on mount (when chunkId set), provide refetch for manual refresh. Use module-level result cache (Map<chunkId, NeighborResult>) to avoid re-fetching the same chunk within a session. Cache is cleared on page navigation or manually via refetch.

This file is new — no existing analog to copy from. Use the same error-handling and async pattern as InspectorRetrieval.tsx (:37-46).
