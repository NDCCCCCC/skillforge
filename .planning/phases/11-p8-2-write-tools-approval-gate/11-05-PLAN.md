type RuleMatcher interface {

// SqlRuleMatcher wraps ApprovalRulesStore; on match, fire-and-forget
// UpdateLastUsedAt via goroutine.
type SqlRuleMatcher struct {
    Store   sqlite.ApprovalRulesStore
    OnMatch func(target, argsHash, skillID, scopeSessionID string)
}

func NewSqlRuleMatcher(store sqlite.ApprovalRulesStore) *SqlRuleMatcher {
    return &SqlRuleMatcher{
        Store: store,
        OnMatch: func(target, argsHash, skillID, scopeSessionID string) {
            _ = store.UpdateLastUsedAt(context.Background(), target, argsHash, skillID, scopeSessionID)
        },
    }
}

func (m *SqlRuleMatcher) Match(ctx context.Context, target, argsHash, skillID, sessionID string) (*sqlite.ApprovalRule, bool) {
    if m == nil || m.Store == nil { return nil, false }
    rule, matched, err := m.Store.MatchRule(ctx, target, argsHash, skillID, sessionID)
    if err != nil || !matched { return nil, false }
    if m.OnMatch != nil && rule != nil {
        go m.OnMatch(rule.Target, rule.ArgsHash, rule.SkillID, rule.ScopeSessionID)
    }
    return rule, true
}

type NoopRuleMatcher struct{}

func (NoopRuleMatcher) Match(_ context.Context, _, _, _, _ string) (*sqlite.ApprovalRule, bool) {
    return nil, false
}

package hitl

import (
    "bytes"
    "crypto/sha256"
    "encoding/hex"
    "encoding/json"
    "errors"
)

// ErrFrozenArgsDrift is returned when the 3-way sha256 byte equality
// assertion fails. The HITL Tool MUST surface this error verbatim.
var ErrFrozenArgsDrift = errors.New("hitl: frozen_args drift detected (preview != audit != executed) — re-interrupt required")

// ComputeFrozenArgsHash returns the hex SHA-256 of the raw bytes.
// D11-D1 rule key + HITL-07 byte-identity anchor.
// NEVER re-marshaled through encoding/json.
func ComputeFrozenArgsHash(args []byte) string {
    h := sha256.Sum256(args)
    return hex.EncodeToString(h[:])
}

// FrozenArgsBytesEqual returns true when two byte slices are byte-identical.
func FrozenArgsBytesEqual(a, b []byte) bool {
    return bytes.Equal(a, b)
}

// assertThreeWayByteEqual checks that the three representations of the
// same frozen_args are byte-identical:
//   previewBytes | auditFrozen | executedBytes
// Returns ErrFrozenArgsDrift on ANY mismatch. HITL-07 invariant.
func assertThreeWayByteEqual(previewBytes, auditFrozen, executedBytes []byte) error {
    if !FrozenArgsBytesEqual(previewBytes, executedBytes) {
        return ErrFrozenArgsDrift
    }
    if !FrozenArgsBytesEqual(auditFrozen, executedBytes) {
        return ErrFrozenArgsDrift
    }
    if !FrozenArgsBytesEqual(previewBytes, auditFrozen) {
        return ErrFrozenArgsDrift
    }
    return nil
}

// VerifyFrozenArgsRoundTrip is a defensive helper for tests + sanity checks.
func VerifyFrozenArgsRoundTrip(raw []byte) ([]byte, error) {
    var any_ any
    if err := json.Unmarshal(raw, &any_); err != nil {
        return nil, err
    }
    return json.Marshal(any_)
}
