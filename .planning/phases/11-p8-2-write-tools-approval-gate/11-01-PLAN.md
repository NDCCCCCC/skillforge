package agent

import (

// RunnerCheckPointID is the canonical format for per-session Runner checkpoint IDs.
// Exported so internal/store/sqlite/checkpoint_test.go::TestCheckpointIDFormat can
// pin it as a regression test. Format: sess:<sessionID>:cp (D11-A1 + D11-C1).
func RunnerCheckPointID(sessionID string) string {
    return "sess:" + sessionID + ":cp"
}

// RunnerBuildConfig is the per-session production configuration for
// adk.NewRunner. Mirrors agent.AgentConfig but switches the agent type
// from react.Agent (legacy Phase 3) to adk.ChatModelAgent (Phase 11).
// Per D11-B4, Agent is per-session singleton; per D11-B1, Tools are
// re-pure-computed per Agent construction.
type RunnerBuildConfig struct {
    Manifest    *skill.Manifest
    Embedder    embedder.Embedder
    Retriever   retriever.Retriever
    ChatModel   model.ToolCallingChatModel
    MaxHops     int                       // 0 → DefaultMaxHops (2)
    SkillID     string                    // for HITL registry target whitelist
    SessionID   string                    // for checkPointID
    Callbacks   []callbacks.Handler        // trust floor + classifier + grounding (D11-A2)
}

// BuildRunner constructs the per-session ChatModelAgent + adk.Runner.
// Returns a *adk.Runner ready for runner.Run(...) or runner.ResumeWithParams(...).
// Per D11-B4: the Agent is per-session singleton (returned Runner is reused
// across all turns of the session). Per D11-B1: the Tool whitelist is
// re-pure-computed every construction as hitl.BuildToolsForSkill(ctx, manifest).
// Read-only Skills get NO write Tools (no HITL); write-Skills get ONLY the
// intersection of AllowedWriteTargets ∩ manifest.Writes.
func BuildRunner(ctx context.Context, cfg RunnerBuildConfig) (*adk.Runner, error) {
    if cfg.Manifest == nil { return nil, errors.New("agent.BuildRunner: Manifest is required") }
    if cfg.ChatModel == nil { return nil, errors.New("agent.BuildRunner: ChatModel is required") }
    if cfg.Embedder == nil { return nil, errors.New("agent.BuildRunner: Embedder is required") }
    if cfg.Retriever == nil { return nil, errors.New("agent.BuildRunner: Retriever is required") }
    if cfg.SessionID == "" { return nil, errors.New("agent.BuildRunner: SessionID is required") }
    if cfg.MaxHops <= 0 { cfg.MaxHops = DefaultMaxHops }

    // Per-turn Tool whitelist (D11-B1 pure function). Empty for Skills
    // with writes:[] — the LLM never sees a write Tool schema for them.
    tools, err := hitl.BuildToolsForSkill(ctx, cfg.Manifest)
    if err != nil { return nil, fmt.Errorf("agent.BuildRunner: BuildToolsForSkill: %w", err) }
    // Always include the read-only retrieve Tool (it is not a write Tool
    // and is not gated by the writes: frontmatter).
    retrieveTool := BuildRetrieveTool(cfg.Embedder, cfg.Retriever, cfg.Manifest)
    tools = append(tools, retrieveTool)

    sysPrompt := systemPrompt(cfg.Manifest)  // from prompts.go

    agent, err := adk.NewChatModelAgent(ctx, &adk.ChatModelAgentConfig{
        Name:        "skillforge-session-" + cfg.SessionID,
        Instruction: sysPrompt,
        Model:       cfg.ChatModel,
        ToolsConfig: adk.ToolsConfig{ToolsConfig: compose.ToolsNodeConfig{Tools: tools}},
        Handlers:    cfg.Callbacks,  // BeforeAgent(trust floor) + BeforeModel(classifier) + AfterModel(grounding)
    })
    if err != nil { return nil, fmt.Errorf("agent.BuildRunner: NewChatModelAgent: %w", err) }

    return adk.NewRunner(ctx, adk.RunnerConfig{
        Agent:           agent,
        EnableStreaming: true,
        // CheckPointStore / CheckPointDeleter are bound by the CALLER via
        // runner.WithCheckPointStore (added in the chatHandler). BuildRunner
        // returns the runner pre-AI; the chatHandler attaches the store.
        // This keeps BuildRunner pure (no store dependency in the agent
        // construction seam) so unit tests can construct Runner without a DB.
    }), nil
}
