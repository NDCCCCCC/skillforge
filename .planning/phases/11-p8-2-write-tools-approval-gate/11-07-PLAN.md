import { useEffect, useState, useCallback } from 'react';
import { Trash2 } from 'lucide-react'; // revoke X icon (lucide-react already loaded per UI-SPEC §Design System)
import { useAppStore } from '../store/app-store';
import { relativeTimeZh } from '../lib/format-effective'; // existing util

export interface ApprovalRuleRow {
  ruleId: string;
  target: string;
  argsHash: string;
  skillId: string;
  scope: 'session' | 'skill';
  scopeSessionId: string;
  createdAt: number;
  lastUsedAt: number | null;
}

const API_BASE = '/api';

async function fetchApprovalRules(
  sessionId: string,
  skillId: string | null,
  fetchImpl: typeof globalThis.fetch = globalThis.fetch,
): Promise<ApprovalRuleRow[] | null> { /* GET /api/session/:id/approval-rules?scope=session|skill&skill_id=...; returns null on error */ }

export default function ApprovalRulesPanel() {
  const activeSessionId = useAppStore((s) => s.activeSessionId);
  const currentSkillId = useAppStore((s) => s.currentSkillId);
  const terminalStatus = useAppStore((s) => s.approval?.terminal?.status ?? null); // 11-06 owns slice; this reads
  const [rules, setRules] = useState<ApprovalRuleRow[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const loadRules = useCallback(async () => {
    setLoading(true);
    setError(null);
    const list = await fetchApprovalRules(activeSessionId, currentSkillId);
    if (list === null) {
      setError('无法加载审批规则。');
      setRules([]);
    } else {
      setRules(list);
    }
    setLoading(false);
  }, [activeSessionId, currentSkillId, terminalStatus]);

  useEffect(() => { void loadRules(); }, [loadRules]);

  // Sort: last_used_at DESC NULLS LAST, created_at DESC
  const sorted = [...rules].sort((a, b) => {
    if (a.lastUsedAt === null && b.lastUsedAt !== null) return 1;
    if (a.lastUsedAt !== null && b.lastUsedAt === null) return -1;
    if (a.lastUsedAt !== b.lastUsedAt) return (b.lastUsedAt ?? 0) - (a.lastUsedAt ?? 0);
    return b.createdAt - a.createdAt;
  });

  return (
    <section data-testid="approval-rules-panel" className="border-b border-border px-5 py-3">
      <h3 className="text-xs font-semibold uppercase tracking-wide text-text-muted">审批规则</h3>

        {loading ? (
          <div data-testid="approval-rules-skeleton" className="text-xs text-text-muted">加载审批规则…</div>
        ) : error ? (
          <div data-testid="approval-rules-error" className="text-xs text-status-non-compliant">{error}</div>
        ) : sorted.length === 0 ? (
          <div data-testid="approval-rules-empty" className="text-xs text-text-muted">
            还没有任何审批规则。批准一次时勾选「记住」即可创建。
          </div>
        ) : (
          sorted.map((rule) => (
            <div
              key={rule.ruleId}
              data-testid="approval-rules-row"
              data-rule-id={rule.ruleId}
              className="group flex items-start justify-between gap-2 rounded-md border border-border bg-surface px-2 py-1.5 hover:bg-surface-2"
            >
              <div className="min-w-0 flex-1">
                <div className="flex items-center gap-1.5">
                  <span className="font-mono text-xs text-text">{rule.target}</span>
                  <span className="text-text-dim">·</span>
                  <span className="text-xs font-semibold text-text-muted">
                    {rule.scope === 'session' ? '本会话' : '整个 Skill'}
                  </span>
                </div>
                <div className="mt-0.5 truncate text-xs text-text-muted">
                  {relativeTimeZh(rule.createdAt)} · 上次使用 {rule.lastUsedAt === null ? '未使用' : relativeTimeZh(rule.lastUsedAt)}

                data-testid="approval-rules-revoke"

                title="撤销规则"
                disabled
                className="flex h-8 w-8 shrink-0 items-center justify-center rounded-md text-text-muted opacity-50"
              >
                <Trash2 className="h-4 w-4" />
              </button>
            </div>
          ))
        )}
      </div>
    </section>
  );
}

async function revokeApprovalRule(
  sessionId: string,
  ruleId: string,
  fetchImpl: typeof globalThis.fetch = globalThis.fetch,
): Promise<{ ok: true } | { ok: false; reason: string }> {
  try {
    const resp = await fetchImpl(
      `${API_BASE}/session/${encodeURIComponent(sessionId)}/approval-rules/${encodeURIComponent(ruleId)}`,
      { method: 'DELETE', headers: { Accept: 'application/json' } },
    );
    if (resp.ok) return { ok: true };
    const reason = resp.status === 404 ? '规则不存在' : `HTTP ${resp.status}`;
    return { ok: false, reason };
  } catch (e) {
    return { ok: false, reason: e instanceof Error ? e.message : '网络错误' };
  }
}

/** One pending approval row from GET /api/session/:id/approval (snake_case wire). */
export interface PendingApprovalRow {
  approval_id: string;
  tool_name: string;
  target: string;
  frozen_args: Record<string, unknown>;
  reason: string;
  created_at: number;
  expires_at: number | null;
  state: 'active' | 'expired';
  skill_version: string;
  prompt_hash: string;
  session_id: string;
  skill_id: string;
  /** When true, an approve-and-remember rule matched; the tool executed without interrupt (HITL-08). */
  auto_passed?: boolean;
}

/** Fetch GET /api/session/:id/approval — returns the pending row, null
 * (no pending), or undefined (error/network). Mirrors the fetchSessions
 * posture: errors degrade to undefined, not throw. */
export async function fetchPendingApproval(
  sessionId: string,
  fetchImpl: typeof globalThis.fetch = globalThis.fetch,
): Promise<PendingApprovalRow | null | undefined> {
  try {
    const resp = await fetchImpl(
      `${API_BASE}/session/${encodeURIComponent(sessionId)}/approval`,
      { headers: { Accept: 'application/json' } },
    );
    if (resp.status === 204) return null; // no pending
    if (!resp.ok) return undefined;
    const body = (await resp.json()) as PendingApprovalRow;
    return body;
  } catch {
    return undefined;
  }
}

import { useEffect } from 'react';
import { useAppStore } from '../store/app-store';
import { fetchPendingApproval, fetchApprovalRules, fetchHistory } from '../lib/sessions';

const RESUME_BANNER_MS = 3000;

export function useResumeBanner(activeSessionId: string): void {
  const setPending = useAppStore((s) => s.setPending);
  const resolvePending = useAppStore((s) => s.resolvePending);
  const clearPending = useAppStore((s) => s.clearPending);
  const hydrateFromGet = useAppStore((s) => s.hydrateFromGet);
  const setResumeBanner = useAppStore((s) => s.setResumeBanner);

  useEffect(() => {
    let cancelled = false;
    let bannerTimer: ReturnType<typeof setTimeout> | null = null;

    void (async () => {
      const [pending, rules, history] = await Promise.all([
        fetchPendingApproval(activeSessionId),
        fetchApprovalRules(activeSessionId, null),
        fetchHistory(activeSessionId), // existing util from chat-restore.ts (rename if needed)
      ]);
      if (cancelled) return;

      // 1) Pending row handling — HITL-03 resume contract
      if (pending === undefined) {
        // network error → don't show banner; ApprovalCard error state surfaces the failure (11-06)
        clearPending();
      } else if (pending === null) {
        clearPending();
      } else if (pending.auto_passed) {
        // hydrateFromGet does NOT handle auto_passed — handle explicitly
        clearPending();
        resolvePending(pending.approval_id, {
          approvalId: pending.approval_id,
          status: 'auto-passed',
          frozenArgs: pending.frozen_args,
          reason: '记忆规则命中',
          createdAt: pending.created_at,
          resolvedAt: Date.now(),
        });
        setResumeBanner({ approvalId: pending.approval_id, status: 'auto-passed', createdAt: Date.now() });
        bannerTimer = setTimeout(() => { if (!cancelled) setResumeBanner(null); }, RESUME_BANNER_MS);
      } else if (pending.state === 'expired') {
        // hydrateFromGet handles the expired branch (terminal + resumeBanner)
        hydrateFromGet(pending);
        bannerTimer = setTimeout(() => { if (!cancelled) setResumeBanner(null); }, RESUME_BANNER_MS);
      } else {
        // pending.state === 'active' → hydrateFromGet does the snake_case→camelCase
        // mapping AND sets pending + resumeBanner (11-06's cross-refresh entry point)
        hydrateFromGet(pending);
        bannerTimer = setTimeout(() => { if (!cancelled) setResumeBanner(null); }, RESUME_BANNER_MS);
      }

      // 2) Rules list hydration (reactive re-fetch on session change)
      // The store does NOT need to own the rules list — ApprovalRulesPanel
      // owns its own useState + useEffect (Task 1+2). This parallel call is
      // for cache warming; the panel re-fetches on its own mount anyway.
      void rules; void history;
    })();

    return () => {
      cancelled = true;
      if (bannerTimer) clearTimeout(bannerTimer);
    };
  }, [activeSessionId, setPending, resolvePending, clearPending, hydrateFromGet, setResumeBanner]);
}
