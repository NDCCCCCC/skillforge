```ts
export const useAppStore = create<State>((set, get) => ({
  // ... existing slices unchanged ...

  approval: {
    pending: null,
    terminal: null,
    history: [],
    resumeBanner: null,

    setPending: (approval) => set((s) => ({
      approval: { ...s.approval, pending: approval, terminal: null },
    })),

    resolvePending: (approvalId, terminal) => set((s) => {
      // Guard: only resolve if pending matches approvalId (defends against stale SSE events)
      if (!s.approval.pending || s.approval.pending.approvalId !== approvalId) {
        return s;  // no-op
      }
      return {
        approval: {
          ...s.approval,
          pending: null,
          terminal,
        },
      };
    }),

    clearPending: () => set((s) => ({ approval: { ...s.approval, pending: null } })),

    clearTerminal: () => set((s) => ({ approval: { ...s.approval, terminal: null } })),

    appendHistory: (entry) => set((s) => ({
      approval: {
        ...s.approval,
        history: [...s.approval.history, entry].slice(-200),  // bounded ring buffer
      },
    })),

    setResumeBanner: (banner) => set((s) => ({
      approval: { ...s.approval, resumeBanner: banner },
    })),

    dismissResumeBanner: () => set((s) => ({
      approval: { ...s.approval, resumeBanner: null },
    })),

    hydrateFromGet: (row) => set((s) => {
      if (row === null) {
        return s;
      }
      if (row.state === 'expired') {
        return {
          approval: {
            ...s.approval,
            terminal: {
              approvalId: row.approvalId,
              status: 'expired',
              frozenArgs: row.frozenArgs,
              reason: 'Skill 或 prompt 已变更，无法恢复',
              createdAt: row.createdAt,
              resolvedAt: Date.now(),
            },
            resumeBanner: {
              approvalId: row.approvalId,
              status: 'expired',
              createdAt: Date.now(),
            },
          },
        };
      }
      return {
        approval: {
          ...s.approval,
          pending: {
            approvalId: row.approvalId,
            toolName: row.toolName,
            target: row.target,
            frozenArgs: row.frozenArgs,
            reason: row.reason || '未提供具体说明',
            createdAt: row.createdAt,
            expiresAt: row.expiresAt,
            skillVersion: row.skillVersion,
            promptHash: row.promptHash,
            sessionId: row.sessionId,
            skillId: row.skillId,
            status: 'pending',
          },
          resumeBanner: {
            approvalId: row.approvalId,
            status: 'pending',
            createdAt: Date.now(),
          },
        },
      };
    }),
  },
}));
```

```tsx
export function ApprovalCard({ sessionId }: { sessionId: string }): JSX.Element | null {
  const pending = useAppStore((s) => s.approval.pending);
  const terminal = useAppStore((s) => s.approval.terminal);

  // Render gate: card visible when pending OR terminal is non-null
  if (pending === null && terminal === null) {
    return null;
  }

  // If pending, render the active interrupt card. If only terminal (e.g. cross-refresh resume shows expired), render terminal state.
  if (pending !== null) {
    return <PendingCardView approval={pending} sessionId={sessionId} />;
  }
  return <TerminalCardView approval={terminal!} />;
}
```
