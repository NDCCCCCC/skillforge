```
// columnExists returns true when the named column is already present in the
// table. Uses PRAGMA table_info() which is read-only and never blocks (the
// migration runs against the DB's own metadata table). Mirrors the
// schema_migrations table pattern used by other Go ORMs but kept local to
// keep the dependency footprint small.
func (s *SQLiteStore) columnExists(table, column string) (bool, error) {
    rows, err := s.db.QueryContext(s.ctx, fmt.Sprintf("PRAGMA table_info(%s)", table))
    if err != nil { return false, fmtError("columnExists", err) }
    defer rows.Close()
    for rows.Next() {
        var cid int
        var name, ctype string
        var notnull, pk int
        var dfltValue sql.NullString
        if err := rows.Scan(&cid, &name, &ctype, &notnull, &dfltValue, &pk); err != nil {
            return false, fmtError("columnExists.Scan", err)
        }
        if name == column { return true, nil }
    }
    return false, rows.Err()
}
```

```
// InsertPendingApproval inserts the FIRST row of the HITL-04 lifecycle
// (action_pending_approval). Called sync by audit_writer.go BEFORE the
// HITL Tool calls tool.StatefulInterrupt (D11-C3 fail-closed). The
// tool_call_id is the PK identity (D11-C4); the partial unique index
// `idx_audit_pending_tcid` rejects duplicate pending rows for the same
// tool_call_id (cross-device race moat).
//
// Returns sql.ErrNoRows-style unique-constraint error on PK collision —
// the caller (audit_writer) treats this as a benign re-call (the second
// POST in the concurrent-approve race finds the row already pending and
// short-circuits).
func (s *SQLiteStore) InsertPendingApproval(ctx context.Context, row PendingApprovalRow) (int64, error) {
    if s.isClosed() { return 0, sql.ErrConnDone }
    res, err := s.db.ExecContext(ctx, putPendingApprovalSQL,
        row.Timestamp.UTC(), row.SessionID, row.UserID, row.SkillID, "action_pending_approval",
        row.Query, row.PromptHash, row.SkillVersion, row.ModelVersion,
        row.VerdictJSON, row.RefusalReason, row.ChunkIDs,
        row.FrozenArgs, row.ToolCallID, row.ApprovalID, row.ScopeSessionID,
    )
    if err != nil {
        return 0, fmtError("InsertPendingApproval", err)
    }
    id, err := res.LastInsertId()
    if err != nil { return 0, fmtError("InsertPendingApproval.LastInsertId", err) }
    return id, nil
}
```

```
// TestConcurrentApproveDenyIdempotency — 10 concurrent INSERTs of
// action_pending_approval with the same tool_call_id → exactly 1 row
// wins (the partial unique index idx_audit_pending_tcid rejects the
// other 9 with UNIQUE constraint error). This is the D11-C4
// cross-device race moat.
func TestConcurrentApproveDenyIdempotency(t *testing.T) {
    dsn := fmt.Sprintf("file:%s?mode=memory&cache=shared&_txlock=immediate", filepath.Join(t.TempDir(), "concurrent.db"))
    ctx := context.Background()
    s, err := sqlite.New(ctx, sqlite.Config{DSN: dsn})
    require.NoError(t, err)
    defer s.Close()

    // Use a barrier to release all goroutines simultaneously.
    var (
        wg        sync.WaitGroup
        start     = make(chan struct{})
        successes int32
        failures  int32
        mu        sync.Mutex
    )

    for i := 0; i < 10; i++ {
        wg.Add(1)
        go func(idx int) {
            defer wg.Done()
            <-start
            _, err := s.InsertPendingApproval(ctx, store.PendingApprovalRow{
                Timestamp:     time.Now().UTC(),
                SessionID:     "sess-race",
                UserID:        "user-1",
                SkillID:       "compliance",
                Query:         "memory:facts",
                PromptHash:    "p-hash",
                SkillVersion:  "v1",
                ModelVersion:  "glm-5.2",
                VerdictJSON:   "",
                RefusalReason: "",
                ChunkIDs:      "",
                FrozenArgs:    []byte(`{"x":1}`),
                ToolCallID:    "tc-race",
                ApprovalID:    "tc-race",
                ScopeSessionID: "sess-race",
            })
            mu.Lock()
            if err == nil {
                successes++
            } else {
                failures++
            }
            mu.Unlock()
        }(i)
    }
    close(start)
    wg.Wait()

    // EXACTLY ONE wins (D11-C4 idempotency moat).
    assert.Equal(t, int32(1), successes, "exactly one of 10 concurrent pending INSERTs must succeed")
    assert.Equal(t, int32(9), failures, "the other 9 must fail with UNIQUE constraint")

    // Verify: exactly 1 pending row exists.
    rows, err := s.QueryAudit(ctx, store.AuditFilter{SessionID: "sess-race", Limit: 10})
    require.NoError(t, err)
    var pendingCount int
    for _, r := range rows {
        if r.Action == "action_pending_approval" && r.ToolCallID == "tc-race" {
            pendingCount++
        }
    }
    assert.Equal(t, 1, pendingCount, "exactly 1 pending row in audit table after concurrent INSERTs")
```
