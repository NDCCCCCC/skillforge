The pure function (D11-B1 — the load-bearing invariant):
```
func BuildToolsForSkill(_ context.Context, manifest *skill.Manifest) ([]tool.BaseTool, error) {
    if manifest == nil { return nil, errors.New("hitl.BuildToolsForSkill: manifest is nil") }
    // D11-B1: per-turn whitelist = intersection(AllowedWriteTargets, manifest.Writes)
    // mapped via hitlHandlers. Read-only Skills (writes:[]) get an empty slice
    // so the LLM NEVER sees a write-tool JSON schema (HITL-05 prompt-injection
    // resistance — structural defense, not policy check).
    writes := manifest.Writes
    if len(writes) == 0 { return nil, nil }

    // Collect (name, tool) pairs so we can sort by tool name without calling
    // Info() twice per pair. ctx is context.Background() here because Info() is
    // required to be stateless by the tool.BaseTool contract (verified at
    // eino@v0.9.14/components/tool/tool.go); real per-Run ctx is not needed.
    type pair struct { name string; t tool.BaseTool }
    pairs := make([]pair, 0, len(writes))
    seenNames := map[string]bool{}
    for _, target := range writes {
        if !contains(AllowedWriteTargets, target) { continue /* unknown target = silently skipped (manifest validator already warned) */ }
        h := LookupHandler(target)
        if h == nil { continue /* handler not yet registered (11-04 review_report) */ }
        info, err := h.Info(context.Background())
        if err != nil { return nil, fmt.Errorf("hitl.BuildToolsForSkill: Tool %s Info: %w", target, err) }
        if seenNames[info.Name] { continue /* duplicate name — first wins */ }
        seenNames[info.Name] = true
        pairs = append(pairs, pair{name: info.Name, t: h})
    }

    // Stable order: sort by tool name so the LLM-visible schema list is
    // deterministic across calls (avoids spurious Tool-call drift in test
    // fixtures; matches the registry ordering convention).
    sort.SliceStable(pairs, func(i, j int) bool { return pairs[i].name < pairs[j].name })
    out := make([]tool.BaseTool, 0, len(pairs))
    for _, p := range pairs { out = append(out, p.t) }
    return out, nil
}
```

```
func BuildToolsForSkill(_ context.Context, manifest *skill.Manifest) ([]tool.BaseTool, error) {
    if manifest == nil { return nil, errors.New("agent.BuildToolsForSkill: manifest is nil") }
    writes := manifest.Writes
    if len(writes) == 0 { return nil, nil }

    type pair struct { name string; t tool.BaseTool }
    pairs := make([]pair, 0, len(writes))
    seenNames := map[string]bool{}
    for _, target := range writes {
        if !hitl.Contains(hitl.AllowedWriteTargets, target) { continue }
        h := hitl.LookupHandler(target)
        if h == nil { continue }
        info, err := h.Info(context.Background())
        if err != nil { return nil, fmt.Errorf("agent.BuildToolsForSkill: Tool %s Info: %w", target, err) }
        if seenNames[info.Name] { continue }
        seenNames[info.Name] = true
        pairs = append(pairs, pair{name: info.Name, t: h})
    }
    sort.SliceStable(pairs, func(i, j int) bool { return pairs[i].name < pairs[j].name })
    out := make([]tool.BaseTool, 0, len(pairs))
    for _, p := range pairs { out = append(out, p.t) }
    return out, nil
}
```
