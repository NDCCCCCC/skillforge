```go
package hitl

const ReportPublishToolName = "report_publish"
const ReportPublishTarget = "report:publish"

type ReportPublishArgs struct {
    Title      string             `json:"title"`
    Period     string             `json:"period"`
    Sections   []SectionDraft     `json:"sections"`    // D18-B4 结构摘要
    KPICallIDs []int64           `json:"kpi_call_ids"` // 审计链关联
}

type SectionDraft struct {
    Title string   `json:"title"`
    KPIs  []string `json:"kpis"` // KPI function names
}

type reportPublishTool struct {
    store     store.Store
    executor  *kpi.Executor
    reportPub func(ctx context.Context, ...) (*report.PublishResult, error) // 注入
}

func (t *reportPublishTool) Info(ctx context.Context) *tool.Info {
    return &tool.Info{
        Name:        ReportPublishToolName,
        Description: "Publish a structured monthly report as a static HTML page",
        Parameters:  reportPublishToolSchema(),
    }
}

func (t *reportPublishTool) Invoke(ctx context.Context, args string) (string, error) {
    var a ReportPublishArgs
    if err := json.Unmarshal([]byte(args), &a); err != nil {
        return "", fmt.Errorf("report_publish: args unmarshal failed: %w", err)
    }

    // === Pass 1: InsertPendingApproval → StatefulInterrupt ===
    isTarget, hasData, data := tool.GetResumeContext[*ApprovalDecision](ctx)
    if !isTarget || !hasData {
        // First invocation: insert pending, interrupt
        approvalID := compose.GetToolCallID(ctx)
        if approvalID == "" {
            approvalID = newApprovalID()
        }
        info := &ApprovalInfo{
            ToolName:        ReportPublishToolName,
            ArgumentsInJSON: args, // frozen_args = 完整参数快照
            ToolCallID:      approvalID,
            Target:          ReportPublishTarget,
        }
        if err := t.auditOrDefault().InsertPendingApproval(ctx, approvalID, info); err != nil {
            return "", fmt.Errorf("report_publish: audit INSERT failed (fail-closed): %w", err)
        }
        return "", tool.StatefulInterrupt(ctx, info, args)
    }

    // === Pass 2: validate decision, then deny / approve / request_change ===
    if err := data.Validate(); err != nil {
        return "", fmt.Errorf("report_publish: invalid ApprovalDecision: %w", err)
    }

    if data.Decision == "deny" {
        return fmt.Sprintf("user denied: %s", data.Reason), nil
    }

    if data.Decision == "request_change" {
        // D18-D4: 要求修改第三态 — 优雅收尾当前 run，chip 预填新 run
        return fmt.Sprintf("change requested: %s", data.Reason), nil
    }

    if data.Decision != "approve" {
        return "", fmt.Errorf("report_publish: unexpected decision: %q", data.Decision)
    }

    // === Approve branch: render + publish ===
    // Reconstruct from frozen args
    var frozen ReportPublishArgs
    if err := json.Unmarshal([]byte(data.FrozenArgs), &frozen); err != nil {
        return "", fmt.Errorf("report_publish: frozen args unmarshal failed: %w", err)
    }

    // Get template from manifest (passed via ctx or闭包)
    tmpl := getManifestTemplate(ctx) // 从 ctx 或 tool 闭包获取

    // Build SlotTexts from Sections
    slots := buildSlotTexts(frozen.Sections)

    // Get KPI snapshots for all KPICallIDs
    snap, err := t.executor.GetSnapshot(ctx, frozen.KPICallIDs)
    if err != nil {
        return "", fmt.Errorf("report_publish: snapshot failed: %w", err)
    }

    // Publish
    approvalAuditID := data.ApprovalAuditID // 从 resume context 获取
    pubResult, err := t.reportPub(ctx, tmpl, slots, snap,
        frozen.Period, t.store, approvalAuditID, frozen.KPICallIDs, frozen.Title)
    if err != nil {
        return "", fmt.Errorf("report_publish: publish failed: %w", err)
    }

    // Return structured success with card data
    card := &wire.ReportCard{
        Title:   frozen.Title,
        Period:  frozen.Period,
        Version: pubResult.Version,
        URL:     pubResult.URL,
    }
    cardBytes, _ := json.Marshal(card)
    return fmt.Sprintf("published: %s", string(cardBytes)), nil
}

func buildSlotTexts(sections []SectionDraft) report.SlotTexts {
    // sections index → slot index → slot text
    // 实际由 LLM 在填写槽位时生成，这里只做占位结构
    slots := make(report.SlotTexts)
    for i := range sections {
        slots[i] = make(map[int]string)
    }
    return slots
}

func getManifestTemplate(ctx context.Context) *report.ReportTemplate {
    // 从 ctx 或闭包获取当前 Skill 的 Template
    // 实现方式：tool 构造时闭包注入
    if tmpl, ok := ctx.Value(templateCtxKey{}).(*report.ReportTemplate); ok {
        return tmpl
    }
    return nil // or fail
}
```

**注意：** reportPublishTool 的 Invoke 是 Pass 1/Pass 2 统一入口，Pass 2 的 frozen args 来自 `data.FrozenArgs`（HITL-04 机制）。approve 分支的 Publish 调用 `internal/report/publish.go` 的 Publish 函数（已在 Plan 01 建立）。

**B. internal/store/sqlite/reports.go — INSERT-only reports 表**

镜像 kpi_calls.go 模式（D16-03-A 先例）：

```go
package sqlite

const putReportSQL = `
INSERT INTO reports (
    ts, period, skill_id, version, title,
    file_path, approval_audit_id, kpi_call_ids, template_version, result_digest
) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
`

const listReportsBaseSQL = `
SELECT id, ts, period, skill_id, version, title,
       file_path, approval_audit_id, kpi_call_ids, template_version, result_digest
FROM reports
`

func (s *SQLiteStore) PutReport(ctx context.Context, row store.ReportRow) error {
    _, err := s.db.ExecContext(ctx, putReportSQL,
        row.TS.UTC().Format(time.RFC3339Nano),
        row.Period, row.SkillID, row.Version, row.Title,
        row.FilePath, row.ApprovalAuditID, row.KPICallIDs,
        row.TemplateVersion, row.ResultDigest,
    )
    if err != nil {
        return fmtError("PutReport", err)
    }
    return nil
}

func (s *SQLiteStore) ListReports(ctx context.Context, period string, limit int) ([]store.ReportRow, error) {
    rows, err := s.db.QueryContext(ctx, listReportsBaseSQL+" WHERE period=? ORDER BY version DESC LIMIT ?", period, limit)
    if err != nil {
        return nil, fmtError("ListReports", err)
    }
    defer rows.Close()
    var reports []store.ReportRow
    for rows.Next() {
        var r store.ReportRow
        // scan...
        reports = append(reports, r)
    }
    return reports, rows.Err()
}

func (s *SQLiteStore) GetReport(ctx context.Context, period string, version int) (*store.ReportRow, error) {
    row := &store.ReportRow{}
    err := s.db.QueryRowContext(ctx,
        listReportsBaseSQL+" WHERE period=? AND version=?",
        period, version).Scan(...)
    if err == sql.ErrNoRows {
        return nil, nil
    }
    if err != nil {
        return nil, fmtError("GetReport", err)
    }
    return row, nil
}
```

**D. cmd/server/main.go — GET /reports/* 静态路由**

```go
import "regexp"

var reportPathAllowlist = regexp.MustCompile(`^[A-Za-z0-9_-]+/v[0-9]+\.html$`)

server.GET("/reports/*filepath", func(ctx context.Context, c *app.RequestContext) {
    p := string(c.Path())
    rel := strings.TrimPrefix(p, "/reports/")
    if !reportPathAllowlist.MatchString(rel) {
        c.SetStatusCode(404)
        return
    }
    file := filepath.Join("reports", filepath.FromSlash(rel))
    b, err := os.ReadFile(file)
    if err != nil {
        c.SetStatusCode(404)
        return
    }
    c.SetContentType("text/html; charset=utf-8")
    c.SetStatusCode(200)
    c.Write(b)
})
```
