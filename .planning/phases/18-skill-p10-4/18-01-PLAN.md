```go
package report

type ReportTemplate struct {
    Title    string            `yaml:"title"    json:"title"`
    Sections []SectionTemplate `yaml:"sections" json:"sections"`
}

type SectionTemplate struct {
    Title string   `yaml:"title" json:"title"`
    KPIs  []string `yaml:"kpis"  json:"kpis"` // KPI function names
    Slots []Slot   `yaml:"slots" json:"slots"`
}

type Slot struct {
    Type string `yaml:"type" json:"type"` // "narrative" | "insight" | "risk_note"
}

// Validate validates the template fail-closed.
// Called at Skill load time via loader.go ValidateAgainstKPI.
func (t *ReportTemplate) Validate(reg *kpi.Registry) error {
    if len(t.Sections) == 0 {
        return fmt.Errorf("template.sections must be non-empty (D18-A4)")
    }
    for i, sec := range t.Sections {
        if strings.TrimSpace(sec.Title) == "" {
            return fmt.Errorf("template.sections[%d].title is empty (D18-A4)", i)
        }
        for _, kpi := range sec.KPIs {
            if _, ok := reg.Lookup(kpi); !ok {
                return fmt.Errorf("template.sections[%d].kpis: unknown KPI %q", i, kpi)
            }
        }
        for j, slot := range sec.Slots {
            if !slices.Contains([]string{"narrative", "insight", "risk_note"}, slot.Type) {
                return fmt.Errorf("template.sections[%d].slots[%d].type %q not in {narrative, insight, risk_note} (D18-A2)", i, j, slot.Type)
            }
        }
    }
    return nil
}
```

**B. internal/report/format.go — KPI 值格式化**

定义 ValueFormatter 类型和实现（Pattern 2 双保险规则）：

```go
package report

// ValueFormatter transforms a KPI numeric value into its final display string.
// Applies 千分位 + 2 decimal places + unit suffix.
// Also applies markdown cell escaping to prevent table injection.
type ValueFormatter func(v any) string

// MarkdownCellEscaper 转义 GFM 表格特殊字符。
var mdCellEscaper = strings.NewReplacer(
    "|", `\|`, "`", "\\`", "*", `\*`, "_", `\_`,
    "\n", " ", "<", `\<`, ">", `\>`,
    "[", `\[`, "]", `\]`,
)

func DefaultValueFormatter(v any) string {
    // 千分位 + 2 位小数 + 单位（从 v 类型推导，或传入显式单位）
    // 只输出 [0-9.,%亿万元-] 白名单字符
    switch val := v.(type) {
    case float64:
        return formatFloat(val)
    case int64:
        return formatInt(val)
    case string:
        // 如果已是格式化字符串，直接转义
        return mdCellEscaper.Replace(val)
    default:
        return mdCellEscaper.Replace(fmt.Sprintf("%v", val))
    }
}

func formatFloat(v float64) string {
    // 千分位 + 2 位小数
    sign := ""
    if v < 0 {
        sign = "-"
        v = -v
    }
    integer := int64(v)
    decimal := (v - float64(integer)) * 100
    return fmt.Sprintf("%s%d.%02d", sign, integer, int64(decimal+0.5))
}

func formatInt(v int64) string {
    // 千分位
    sign := ""
    if v < 0 {
        sign = "-"
        v = -v
    }
    s := fmt.Sprintf("%d", v)
    var result strings.Builder
    for i, r := range s {
        if i > 0 && (len(s)-i)%3 == 0 {
            result.WriteByte(',')
        }
        result.WriteRune(r)
    }
    return sign + result.String()
}
```

```go
package report

import (
    "bytes"
    "strings"

    "github.com/yuin/goldmark"
    "github.com/yuin/goldmark/extension"
    "github.com/microcosm-cc/bluemonday"
)

var md = goldmark.New(
    goldmark.WithExtensions(extension.GFM, extension.CJK),
    // 刻意不传 html.WithUnsafe() — raw HTML 默认转义是第一层防御
)

var policy = sync.OnceValue(func() *bluemonday.Policy {
    p := bluemonday.UGCPolicy()
    p.AllowElements("section", "table", "thead", "tbody", "tr", "th", "td")
    p.AllowTables() // policies.go:217 源码核实
    return p
})

// SlotTexts maps section index + slot index to LLM-generated text.
type SlotTexts map[int]map[int]string // sectionIdx -> slotIdx -> text

// Render executes the sandcastle pipeline: bind → goldmark → bluemonday.
// Order is STRUCTURALLY LOCKED — TestRenderPipelineOrder asserts this sequence.
func Render(tmpl *ReportTemplate, slots SlotTexts, snap *kpi.Snapshot, fmtFn ValueFormatter) ([]byte, error) {
    body := buildMarkdown(tmpl, slots, snap, fmtFn) // 1. 绑定：数据绑定，LLM 文字原样，KPI 值已格式化+转义
    var buf bytes.Buffer
    if err := md.Convert([]byte(body), &buf); err != nil { // 2. goldmark: raw HTML 转义
        return nil, err
    }
    return policy().SanitizeBytes(buf.Bytes()), nil // 3. bluemonday: 永远最后
}

// buildMarkdown assembles the markdown body from template + slot texts + KPI values.
// LLM-generated slot text is inserted verbatim (text/template 不转义).
// KPI values are formatted by fmtFn before insertion.
func buildMarkdown(tmpl *ReportTemplate, slots SlotTexts, snap *kpi.Snapshot, fmtFn ValueFormatter) string {
    var sb strings.Builder
    sb.WriteString("# ")
    sb.WriteString(tmpl.Title)
    sb.WriteString("\n\n")

    for si, sec := range tmpl.Sections {
        sb.WriteString("## ")
        sb.WriteString(sec.Title)
        sb.WriteString("\n")

        // KPI table
        sb.WriteString("| 指标 | 值 |\n|---|---|\n")
        for _, kpi := range sec.KPIs {
            val := snap.Get(kpi) // *kpi.Value
            display := fmtFn(val.Interface())
            sb.WriteString("| ")
            sb.WriteString(mdCellEscaper.Replace(kpi))
            sb.WriteString(" | ")
            sb.WriteString(display)
            sb.WriteString(" |\n")
        }
        sb.WriteString("\n")

        // Slot texts
        if slotMap, ok := slots[si]; ok {
            for slotIdx, text := range slotMap {
                slotType := sec.Slots[slotIdx].Type
                sb.WriteString("### ")
                sb.WriteString(slotType)
                sb.WriteString("\n\n")
                sb.WriteString(text)
                sb.WriteString("\n\n")
            }
        }
    }
    return sb.String()
}
```

```go
// PublishResult contains the output of a successful publish.
type PublishResult struct {
    Version   int
    FilePath  string
    URL       string // /reports/<period>/v<N>.html
}

var reportPathRe = regexp.MustCompile(`^[A-Za-z0-9_-]+/v[0-9]+\.html$`)

// Publish renders the report and writes it to reports/<period>/v<N>.html.
// vN increments atomically within the same period.
func Publish(ctx context.Context, tmpl *ReportTemplate, slots SlotTexts, snap *kpi.Snapshot,
    period string, store store.Store, approvalAuditID string, kpiCallIDs []int64,
    skillID string) (*PublishResult, error) {

    // 1. Render
    html, err := Render(tmpl, slots, snap, DefaultValueFormatter)
    if err != nil {
        return nil, fmt.Errorf("report: render failed: %w", err)
    }

    // 2. vN increment
    version, err := getNextVersion(ctx, store, period)
    if err != nil {
        return nil, fmt.Errorf("report: version increment failed: %w", err)
    }

    // 3. Write file
    dir := filepath.Join("reports", period)
    if err := os.MkdirAll(dir, 0755); err != nil {
        return nil, fmt.Errorf("report: mkdir failed: %w", err)
    }
    filename := fmt.Sprintf("v%d.html", version)
    filePath := filepath.Join(dir, filename)

    if err := os.WriteFile(filePath, html, 0644); err != nil {
        return nil, fmt.Errorf("report: write failed: %w", err)
    }

    // 4. Digest
    digest := sha256Hex(html)

    // 5. PutReport
    templateVersion := templateHash(tmpl)
    row := store.ReportRow{
        Period:           period,
        SkillID:          skillID,
        Version:          version,
        Title:            tmpl.Title,
        FilePath:         filePath,
        ApprovalAuditID:  approvalAuditID,
        KPICallIDs:       kpiCallIDs,
        TemplateVersion:   templateVersion,
        ResultDigest:     digest,

    return &PublishResult{
        Version:  version,
        FilePath: filePath,
        URL:      "/reports/" + period + "/" + filename,

}
```
