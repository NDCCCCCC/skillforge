```go
// FanOutDispatchArgsV2 is the data-channel fan-out classifier args.
// D18-C1: per-章节 boundary 由模板定义; D18-C2: 主循环 tool-call 自决。
type FanOutDispatchArgsV2 struct {
    ShouldFanOut   bool     `json:"should_fan_out"`
    NumSubtasks    int      `json:"num_subtasks"`
    TaskBoundaries []string `json:"task_boundaries"` // 模板 section title 逐字匹配
    Reason         string   `json:"reason"`           // 分类理由 — 进审计行 (FAN-05)
}

// fallbackReason enumerations (D18-C4)
const (
    FallbackInvalidJSON            = "invalid_json"
    FallbackMissingField           = "missing_field"
    FallbackBoundaryNotInTemplate  = "boundary_not_in_template"
    FallbackNumSubtasksMismatch    = "num_subtasks_mismatch"
)

// validateTaskBoundaries checks that every boundary exists in template section titles.
func validateTaskBoundaries(boundaries []string, sectionTitles []string) (string, bool) {
    allowed := make(map[string]bool)
    for _, t := range sectionTitles {
        allowed[t] = true
    }
    for _, b := range boundaries {
        if !allowed[b] {
            return b, false
        }
    }
    return "", true
}
```

**实现 InvokableRun 的数据通道变体：**

```go
func (t *dataChannelFanOutTool) Invoke(ctx context.Context, args string) (string, error) {
    // 1. 解析 args（JSON → FanOutDispatchArgsV2）
    var a FanOutDispatchArgsV2
    if err := json.Unmarshal([]byte(args), &a); err != nil {
        // invalid JSON → fallback
        return `{"status":"fallback_sequential","reason":"invalid_json"}`, nil
    }

    // 2. 字段存在性校验
    if a.TaskBoundaries == nil {
        return `{"status":"fallback_sequential","reason":"missing_field"}`, nil
    }

    // 3. task_boundaries 闭包校验
    if _, ok := validateTaskBoundaries(a.TaskBoundaries, t.sectionTitles); !ok {
        return fmt.Sprintf(
            `{"status":"fallback_sequential","reason":"boundary_not_in_template","detail":"%s"}`,
            "task_boundary not in template sections",
        ), nil
    }

    // 4. num_subtasks 校验
    if a.NumSubtasks != len(a.TaskBoundaries) {
        return `{"status":"fallback_sequential","reason":"num_subtasks_mismatch"}`, nil
    }

    // 5. 校验通过：fan-out 执行
    if a.ShouldFanOut {
        // errgroup 并行取数
        g, ctx := errgroup.WithContext(ctx)
        g.SetLimit(4)
        results := make([]*kpi.Value, len(a.TaskBoundaries))

        for i, sectionTitle := range a.TaskBoundaries {
            sectionIdx := i // capture
            g.Go(func() error {
                // 找到 section 对应的 KPIs
                kpis := t.sectionKPIs[sectionIdx]
                var snap []*kpi.Value
                for _, kpi := range kpis {
                    v, err := t.executor.Call(ctx, kpi, t.period)
                    if err != nil {
                        return err
                    }
                    snap = append(snap, v)
                }
                results[sectionIdx] = mergeSnapshot(snap) // 合并为 section snapshot
                return nil
            })
        }
        if err := g.Wait(); err != nil {
            return "", err
        }
        // 构建结果 JSON
        result, _ := json.Marshal(map[string]interface{}{
            "status":        "fan_out_success",
            "num_subtasks":  a.NumSubtasks,
            "boundaries":    a.TaskBoundaries,
            "reason":        a.Reason,
        })
        return string(result), nil
    }

    // 6. should_fan_out=false → 顺序执行（fallback）
    return `{"status":"fallback_sequential","reason":"classifier_declined_fan_out"}`, nil
}
```

**fallback 行为：** 返回非 error 的结构化 JSON string，LLM 收到后继续在主循环中顺序发 kpi_call 调用（按章节顺序，逐章节调用）。这是 D18-C4 规定的 fallback 策略。

```tsx
const handleRequestChange = useCallback(async () => {
  const reason = requestChangeReason.trim();
  if (reason.length < 3) return;
  setInFlight('request_change');
  try {
    const status = await postDecision(sessionId, approval.approvalId, {
      decision: 'request_change',
      reason,
      target: approval.target,
      approval_id: approval.approvalId,
    });
    setTerminalStatus('change_requested');
  } catch (e) {
    // error handling
  } finally {
    setInFlight(null);
    setRequestChangeOpen(false);
  }
}, [requestChangeReason, sessionId, approval]);
```
