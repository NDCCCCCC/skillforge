Component structure:
```tsx
interface DocUploadProps {
  // No props needed — panel is self-contained
}

type UploadState = 'idle' | 'uploading' | 'parsing' | 'indexing' | 'done' | 'error'

export default function DocUpload() {
  const [state, setState] = useState<UploadState>('idle')
  const [bases, setBases] = useState<KBEntry[]>([])  // from GET /api/knowledge/bases
  const [selectedSlug, setSelectedSlug] = useState('')
  const [file, setFile] = useState<File | null>(null)
  const [error, setError] = useState('')
  const [chunks, setChunks] = useState<number | null>(null)  // for done state

  // Load KB list on mount
  useEffect(() => {
    fetch('/api/knowledge/bases')
      .then(r => r.json())
      .then((d: { bases: KBEntry[] }) => setBases(d.bases))
      .catch(() => {})  // non-critical
  }, [])

  function handleFileChange(e: React.ChangeEvent<HTMLInputElement>) {
    setFile(e.target.files?.[0] ?? null)
    setError('')
  }

  async function handleSubmit() {
    if (!selectedSlug || !file) return
    setError('')
    setState('uploading')

    const formData = new FormData()
    formData.append('file', file)
    // Optional metadata JSON
    formData.append('metadata', JSON.stringify({}))

    const token = localStorage.getItem('sf_admin_token') ?? ''
    try {
      const resp = await fetch(`/api/admin/kb/${encodeURIComponent(selectedSlug)}/upload`, {
        method: 'POST',
        headers: { 'X-Admin-Token': token },
        body: formData,
      })

      if (resp.ok) {
        const data = await resp.json() as { chunks: number }
        setChunks(data.chunks)
        setState('done')
      } else {
        const body = (await resp.json().catch(() => ({}))) as { error?: string }
        setError(body.error ?? `上传失败（HTTP ${resp.status}）`)
        setState('error')
      }
    } catch (e) {
      setError('网络错误')
      setState('error')
    }
  }

  // State machine rendering:
  // idle: "上传文档" button (expandable section)
  // uploading: progress bar + "正在上传..."
  // parsing: spinner + "正在解析文档..."
  // indexing: spinner + "正在索引入库..."
  // done: green badge + "{chunks} 个片段已入库"
  // error: red alert + error text verbatim
}
```

1. KB dropdown (active slugs only from GET /api/knowledge/bases)
2. File picker (accept=".md,.pdf,.docx,.xlsx,.pptx")
3. Submit button
4. State rendering below

The KB dropdown only shows active slugs (bases.filter(b => true)). The existing KnowledgeBaseList already lists all slugs with active/inactive state — DocUpload only needs active slugs for the dropdown.

For the progress tracking (uploading/parsing/indexing states): the XHR `upload.onprogress` gives upload progress. The parsing/indexing states are server-side and shown as a spinner after the upload completes. The FE cannot know when parsing/indexing starts/ends without SSE — use a fixed sequence: once the HTTP request is sent, show "uploading" until it finishes, then "parsing" briefly (200ms minimum so user sees it), then "indexing", then "done" or "error".

<pitfalls>
## Pitfalls to Avoid

**No new route for DocUpload.** It is a sub-panel in the /admin view, not a standalone page. The state machine lives in the component.

**X-Admin-Token must be sent on every fetch.** The same pattern as KnowledgeBaseList: `headers: { 'X-Admin-Token': token }`. If missing, the server returns 403.

**Backend error text is rendered verbatim.** Do not sanitize or transform the error message. `body.error ?? `上传失败（HTTP ${resp.status}）`` is the fallback.

**KBM-06: deactivated slugs are NOT in GET /api/knowledge/bases.** The public projection only returns active slugs. This is correct — you can only upload to an active KB.
</pitfalls>
