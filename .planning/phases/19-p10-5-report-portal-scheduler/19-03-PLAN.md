Create internal/scheduler/cleanup_job.go.

TTL cleanup job deletes anonymized DuckDB dataset files whose time-to-live has expired. Only anonymized datasets (datasets.tier == "anonymized") are eligible. Audit rows and KPI snapshots are NEVER deleted.

```go
package scheduler

import (
    "context"
    "log"
    "os"
    "path/filepath"
    "time"

    "github.com/skillforge/skillforge/internal/store"
)

// TTLCleanupConfig configures the TTL cleanup job.
type TTLCleanupConfig struct {
    // TTL is how old a dataset must be before it is cleaned up.
    // Datasets created before (now - TTL) are eligible.
    TTL time.Duration
    // DatasetsDir is the directory containing .duckdb files.
    DatasetsDir string
}

// RunTTLCleanup scans the datasets table for anonymized datasets older than TTL
// and deletes their files after DETACHing from the DuckDB engine.
func RunTTLCleanup(ctx context.Context, store store.Store, lru interface {
    Detach(ctx context.Context, id string) error
}, cfg TTLCleanupConfig) error {
    // List all datasets
    // For each dataset where tier == "anonymized" and created_at < now - TTL:
    //   1. DETACH via lru.Detach(ctx, dataset.Name)
    //   2. os.Remove(dataset.FilePath)
    //   3. Log the deletion
    // Audit rows (kpi_calls, report_runs, audit) are NEVER touched.
    return nil
}
```

Key implementation details:

1. Query the store for all datasets with `tier == "anonymized"`. The store needs a new method `ListDatasets(ctx)` (add it to the Store interface and sqlite implementation).

2. For each dataset, check if `created_at < now - TTL`. The TTL default should be configurable but a reasonable default is 90 days.

3. Before deleting the file, call `lru.Detach(ctx, dataset.Name)` to release the DuckDB file handle. This is CRITICAL on Windows — without DETACH, the file is locked and `os.Remove` fails silently.

4. Call `os.Remove(dataset.FilePath)` only after successful DETACH.

5. Log each deletion with dataset name, age, and file path.

6. If DETACH fails, log the error and skip the file (do not attempt to delete).

7. If the file is already gone (stray registry entry), log a warning and continue.

8. The cleanup is idempotent — running it twice on the same dataset is safe (second run finds no file, logs a warning, continues).

Add to the Scheduler in scheduler.go:
```go
// In Start(), after registering report jobs:
if cfg.TTLCleanupSchedule != "" {
    _, err := s.cron.AddFunc(cfg.TTLCleanupSchedule, func() {
        s.runTTLCleanup(ctx)
    })
    // ...
}
```

The TTL cleanup schedule should be configurable via a global config or environment variable. Default: `"0 2 * * *"` (daily at 2am).

The `store.Store` interface needs a new method:
```go
// ListDatasets returns all registered datasets.
ListDatasets(ctx context.Context) ([]DatasetRow, error)
```

Add this to store.go interface and sqlite/datasets.go implementation.

The portal shows all report runs in a table with filtering and manual trigger capability.

```tsx
import { useState, useEffect } from 'react';
import { useStore } from '../store/app-store';

interface ReportRun {
  run_id: string;
  skill_id: string;
  period: string;
  cron_expr: string;
  fired_at: string;
  kpi_snapshots: string;
  output_url: string;
  status: 'success' | 'failed' | 'skipped_duplicate' | 'eval_gate_blocked';
  drift_detected: boolean;
  created_at: string;
}

interface ReportPortalProps {
  // skill_id and period come from URL params or filter state
}

export function ReportPortal() {
  const [runs, setRuns] = useState<ReportRun[]>([]);
  const [loading, setLoading] = useState(false);
  const [skillFilter, setSkillFilter] = useState('');
  const [periodFilter, setPeriodFilter] = useState('');

  const fetchRuns = async () => {
    setLoading(true);
    try {
      const params = new URLSearchParams();
      if (skillFilter) params.set('skill_id', skillFilter);
      if (periodFilter) params.set('period', periodFilter);
      const res = await fetch(`/api/reports?${params}`);
      const data = await res.json();
      setRuns(Array.isArray(data) ? data : []);
    } catch (e) {
      console.error('Failed to fetch reports:', e);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => { fetchRuns(); }, [skillFilter, periodFilter]);

  const handleRerun = async (skillId: string, period: string) => {
    const res = await fetch('/api/reports/rerender', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ skill_id: skillId, period }),
    });
    if (res.ok) {
      fetchRuns(); // refresh list
    }
  };

  return (
    <div className="report-portal">
      <h2>Report Runs</h2>
      <div className="filters">
        <input
          placeholder="Skill ID"
          value={skillFilter}
          onChange={e => setSkillFilter(e.target.value)}
        />
        <input
          placeholder="Period (e.g. 2026-08)"
          value={periodFilter}
          onChange={e => setPeriodFilter(e.target.value)}
        />
        <button onClick={fetchRuns}>Filter</button>
      </div>
      {loading ? (
        <p>Loading...</p>
      ) : (
        <table>
          <thead>
            <tr>
              <th>Skill</th>
              <th>Period</th>
              <th>Status</th>
              <th>Fired At</th>
              <th>Drift</th>
              <th>Actions</th>
            </tr>
          </thead>
          <tbody>
            {runs.map(run => (
              <tr key={run.run_id}>
                <td>{run.skill_id}</td>
                <td>{run.period}</td>
                <td>
                  <span className={`status-badge ${run.status}`}>
                    {run.status}
                  </span>
                </td>
                <td>{new Date(run.fired_at).toLocaleString()}</td>
                <td>
                  {run.drift_detected ? (
                    <span className="drift-badge" title="KPI values changed beyond threshold">
                      DRIFT
                    </span>
                  ) : '—'}
                </td>
                <td>
                  {run.output_url && (
                    <a href={run.output_url} target="_blank" rel="noopener noreferrer">
                      View Report
                    </a>
                  )}
                  <button onClick={() => handleRerun(run.skill_id, run.period)}>
                    Rerun
                  </button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      )}
    </div>
  );
}
```
