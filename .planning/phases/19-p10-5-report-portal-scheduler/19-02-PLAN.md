This module checks the eval gate before a cron-triggered report run fires (GEVL-11).

The scheduler needs to know the latest faithfulness score for a Skill. Since the eval store (internal/eval) uses FilesystemEvalReporter that writes JSON reports to eval/reports/, the scheduler needs to scan those reports to find the latest per-skill faithfulness.

```go
package scheduler

import (
    "encoding/json"
    "fmt"
    "os"
    "path/filepath"
    "sort"
    "time"

    "github.com/skillforge/skillforge/internal/eval"
)

// EvalGate checks whether a Skill's latest eval faithfulness score is above
// the threshold. It scans the eval/reports directory for the Skill's most recent
// report and extracts its faithfulness value.
type EvalGate struct {
    evalReportsDir string // e.g. "eval/reports"
    thresholdsDir  string // e.g. "skills"
}

// NewEvalGate constructs an EvalGate with the given directories.
func NewEvalGate(evalReportsDir, thresholdsDir string) *EvalGate {
    return &EvalGate{evalReportsDir: evalReportsDir, thresholdsDir: thresholdsDir}
}

// Check checks the eval gate for skillID. Returns (true, nil) if the skill passes
// (faithfulness >= threshold or no eval data). Returns (false, reason) if blocked.
func (g *EvalGate) Check(skillID string) (bool, string, error) {
    // Load thresholds for this skill
    threshold, err := g.loadThreshold(skillID)
    if err != nil {
        return false, "", fmt.Errorf("eval gate: load threshold: %w", err)
    }

    // Get latest faithfulness for this skill
    score, err := g.getLatestFaithfulness(skillID)
    if err != nil {
        // No eval data — allow the run (fail open for v1; eval may not have run yet)
        return true, "no_eval_data", nil
    }

    if score < threshold {
        return false, fmt.Sprintf("faithfulness %.4f < threshold %.4f", score, threshold), nil
    }
    return true, "", nil
}

func (g *EvalGate) loadThreshold(skillID string) (float64, error) {
    // Try skill-specific eval-gate.json first
    gatePath := filepath.Join(g.thresholdsDir, skillID, "eval-gate.json")
    b, err := os.ReadFile(gatePath)
    if err == nil {
        var cfg struct {
            FaithfulnessThreshold *float64 `json:"faithfulness_threshold"`
        }
        if err := json.Unmarshal(b, &cfg); err == nil && cfg.FaithfulnessThreshold != nil {
            return *cfg.FaithfulnessThreshold, nil
        }
    }
    // Fall back to default
    return eval.DefaultFaithfulnessThreshold, nil
}

func (g *EvalGate) getLatestFaithfulness(skillID string) (float64, error) {
    entries, err := os.ReadDir(g.evalReportsDir)
    if err != nil {
        return 0, fmt.Errorf("read eval reports dir: %w", err)
    }

    var latest os.FileInfo
    var latestPath string
    for _, e := range entries {
        if e.IsDir() || filepath.Ext(e.Name()) != ".json" {
            continue
        }
        // Look for reports that mention this skill
        // Report filenames are timestamps: 2026-08-28T09-00-00.json
        if latest == nil || e.ModTime().After(latest.ModTime()) {
            latest = e
            latestPath = filepath.Join(g.evalReportsDir, e.Name())
        }
    }

    if latestPath == "" {
        return 0, fmt.Errorf("no eval reports found")
    }

    b, err := os.ReadFile(latestPath)
    if err != nil {
        return 0, fmt.Errorf("read report: %w", err)
    }

    var report struct {
        Skill        string             `json:"skill"`
        Overall      map[string]float64 `json:"overall"`
    }
    if err := json.Unmarshal(b, &report); err != nil {
        return 0, fmt.Errorf("parse report: %w", err)
    }

    if report.Skill != skillID {
        // Not the right skill, find the right one
        return g.findSkillFaithfulness(skillID)
    }

   faith, ok := report.Overall["faithfulness"]
    if !ok {
        return 0, fmt.Errorf("no faithfulness in report")
    }
    return faith, nil
}

// findSkillFaithfulness scans all reports to find the latest for skillID.
func (g *EvalGate) findSkillFaithfulness(skillID string) (float64, error) {
    entries, err := os.ReadDir(g.evalReportsDir)
    if err != nil {
        return 0, err
    }

    type candidate struct {
        faith float64
        time  time.Time
    }
    var best candidate

    for _, e := range entries {
        if e.IsDir() || filepath.Ext(e.Name()) != ".json" {
            continue
        }
        path := filepath.Join(g.evalReportsDir, e.Name())
        b, err := os.ReadFile(path)
        if err != nil {
            continue
        }
        var report struct {
            Skill   string             `json:"skill"`
            Overall map[string]float64 `json:"overall"`
        }
        if err := json.Unmarshal(b, &report); err != nil {
            continue
        }
        if report.Skill != skillID {
            continue
        }
        faith, ok := report.Overall["faithfulness"]
        if !ok {
            continue
        }
        t := e.ModTime()
        if best.time.IsZero() || t.After(best.time) {
            best = candidate{faith: faith, time: t}
        }
    }

    if best.time.IsZero() {
        return 0, fmt.Errorf("no faithfulness for skill %s", skillID)
    }
    return best.faith, nil
}
```

Drift detection compares KPI snapshots between consecutive runs. A drift is detected when any KPI value changes by more than the threshold (default 5%).

```go
package scheduler

import (
    "encoding/json"
    "fmt"
    "math"
)

// DetectDrift compares previous and current KPI snapshots and returns true if
// any KPI value differs by more than the configured threshold. The prev map
// may be nil (first run — no drift possible).
func DetectDrift(prev, curr map[string]any, threshold float64) (bool, error) {
    if len(prev) == 0 {
        return false, nil // first run — no drift possible
    }
    if len(curr) == 0 {
        return false, fmt.Errorf("current snapshot is empty")
    }
    for kpi, prevVal := range prev {
        currVal, ok := curr[kpi]
        if !ok {
            // KPI present in previous but not current — not a drift, just a new KPI set
            continue
        }
        prevF, currF, ok := toFloat64(prevVal), toFloat64(currVal), true
        if !ok {
            // Non-numeric values — skip
            continue
        }
        if prevF == 0 {
            // Cannot compute relative delta for zero baseline — skip
            continue
        }
        delta := math.Abs((currF - prevF) / prevF)
        if delta > threshold {
            return true, nil
        }
    }
    return false, nil
}

// toFloat64 converts a numeric any to float64. Returns ok=false if the value
// is not a numeric type (int, float, uint).
func toFloat64(v any) (float64, bool) {
    switch n := v.(type) {
    case float64:
        return n, true
    case float32:
        return float64(n), true
    case int:
        return float64(n), true
    case int64:
        return float64(n), true
    case int32:
        return float64(n), true
    case uint64:
        return float64(n), true
    case uint32:
        return float64(n), true
    default:
        return 0, false
    }
}

// ParseKPISnapshots parses a JSON KPI snapshot string into a map.
// Used by the store layer to deserialize report_runs.kpi_snapshots.
func ParseKPISnapshots(raw string) (map[string]any, error) {
    if raw == "" {
        return nil, nil
    }
    var m map[string]any
    if err := json.Unmarshal([]byte(raw), &m); err != nil {
        return nil, fmt.Errorf("parse kpi snapshots: %w", err)
    }
    return m, nil
}

// SerializeKPISnapshots serializes a KPI snapshot map to JSON.
func SerializeKPISnapshots(m map[string]any) (string, error) {
    if m == nil {
        return "", nil
    }
    b, err := json.Marshal(m)
    if err != nil {
        return "", fmt.Errorf("serialize kpi snapshots: %w", err)
    }
    return string(b), nil
}
```

Key behaviors:
- First run (prev == nil or empty) always returns `drift_detected = false`
- Only per-KPI relative delta is compared (not absolute)
- Zero baseline values are skipped (cannot detect relative drift from zero)
- Non-numeric values are skipped
- Threshold is configurable per-Skill via manifest or global default (0.05 = 5%)

```go
package scheduler

import (
    "context"
    "fmt"
    "log"
    "time"

    "github.com/google/uuid"
    "github.com/skillforge/skillforge/internal/skill"
    "github.com/skillforge/skillforge/internal/store"
)

// ReportJobInput is the input for triggering a report run (from cron or manual trigger).
type ReportJobInput struct {
    SkillID  string
    Period   string
    CronExpr string
    IsManual bool // true = manual trigger, false = cron trigger
}

// RunReportJob executes a single report run for the given skill and period.
// It checks the eval gate, acquires the in-flight slot, runs the DeepAgent path,
// captures KPI snapshots, detects drift, and writes the report_runs audit row.
func RunReportJob(ctx context.Context, input ReportJobInput, store store.Store, evalGate *EvalGate) error {
    key := SkillPeriodKey(input.SkillID, input.Period)

    // 1. Eval gate check (GEVL-11) — skip cron runs if faithfulness regressed
    if !input.IsManual {
        pass, reason, err := evalGate.Check(input.SkillID)
        if err != nil {
            log.Printf("[WARN] scheduler: eval gate check failed for %s: %v", key, err)
            // Fail open: log and proceed
        } else if !pass {
            log.Printf("[INFO] scheduler: eval gate blocked %s: %s", key, reason)
            writeBlockedRun(ctx, input, store, "eval_gate_blocked", reason)
            return nil
        }
    }

    // 2. Concurrent guard — acquire in-flight slot
    if !tryAcquireInFlight(key) {
        log.Printf("[INFO] scheduler: skipping %s (already in flight)", key)
        writeBlockedRun(ctx, input, store, "skipped_duplicate", "concurrent run in progress")
        return nil
    }
    defer releaseInFlight(key)

    // 3. Get previous KPI snapshot for drift detection
    prevRow, err := store.GetLatestReportRunBySkill(ctx, input.SkillID, input.Period)
    var prevSnapshots map[string]any
    if err != nil {
        log.Printf("[WARN] scheduler: GetLatestReportRunBySkill failed: %v", err)
    } else if prevRow != nil {
        prevSnapshots, _ = ParseKPISnapshots(prevRow.KPISnapshots)
    }

    // 4. Run the DeepAgent report generation path
    // TODO: This calls the same DeepAgent path that HITL report_publish uses.
    // The runID is a new UUID for this run.
    runID := uuid.New().String()
    firedAt := time.Now().UTC()

    // Collect KPI snapshots by calling the KPI executor (same as kpi_call tool)
    currSnapshots, runErr := executeReportRun(ctx, input.SkillID, input.Period)

    // 5. Detect drift
    driftDetected := false
    if runErr == nil && len(currSnapshots) > 0 {
        driftDetected, _ = DetectDrift(prevSnapshots, currSnapshots, 0.05) // 5% default threshold
    }

    // 6. Determine status
    status := "success"
    outputURL := ""
    if runErr != nil {
        status = "failed"
        log.Printf("[ERROR] scheduler: report run failed for %s: %v", key, runErr)
    } else {
        // Run succeeded — publish the report (reuse report.Publish)
        // outputURL = "/reports/period/vN.html"
        // report.Publish handles version increment atomically
    }

    // 7. Write report_runs audit row
    snapshotJSON, _ := SerializeKPISnapshots(currSnapshots)
    row := store.ReportRunRow{
        RunID:        runID,
        SkillID:      input.SkillID,
        Period:       input.Period,
        CronExpr:     input.CronExpr,
        FiredAt:      firedAt,
        KPISnapshots: snapshotJSON,
        OutputURL:    outputURL,
        Status:       status,
        DriftDetected: driftDetected,
        CreatedAt:    time.Now().UTC(),
    }
    if err := store.PutReportRun(ctx, row); err != nil {
        log.Printf("[ERROR] scheduler: PutReportRun failed: %v", err)
    }

    return runErr
}

// writeBlockedRun writes a report_runs row for a blocked/skipped run.
func writeBlockedRun(ctx context.Context, input ReportJobInput, store store.Store, status, reason string) {
    row := store.ReportRunRow{
        RunID:       uuid.New().String(),
        SkillID:     input.SkillID,
        Period:      input.Period,
        CronExpr:    input.CronExpr,
        FiredAt:     time.Now().UTC(),
        Status:      status,
        DriftDetected: false,
        CreatedAt:   time.Now().UTC(),
    }
    // Add reason to KPISnapshots as a note
    row.KPISnapshots, _ = SerializeKPISnapshots(map[string]any{"skip_reason": reason})
    _ = store.PutReportRun(ctx, row)
}

// executeReportRun calls the KPI executor to get current KPI values.
// This is the same kpi_call logic used by the DeepAgent during report generation.
// Returns a map of KPI name -> value.
func executeReportRun(ctx context.Context, skillID, period string) (map[string]any, error) {
    // This stub will be implemented by calling the KPI executor.

    // internal/kpi.Executor.Call or internal/agent/kpi_admin.go.
    // For now return an empty map; the full integration with KPI executor
    // will be wired in 19-02.

}
```
