package kb

// Role 是封闭的四值语义角色词表（KBM-03「四选一」）。角色是平台语义不是配置——
// 这 4 个常量是全仓唯一允许 hard-code 的 KB 词汇（26-10 结构门白名单）。
type Role string

const (
    RoleAuthority  Role = "authority"   // 唯一可引用证据源（citation 白名单仅此角色）
    RoleHistorical Role = "historical"  // 被取代版本留档；显式 reads 才可达；不可引用
    RolePersonal   Role = "personal"    // 用户私有；默认排除检索，显式声明才可读；不可引用
    RoleDraft      Role = "draft"       // 未生效；永不进检索，仅入库预览；不可引用
)

func Roles() []Role            // 4 值确定性序
func IsRole(s string) bool     // 封闭集判定（seed/端点校验用）
const CitableRole = RoleAuthority  // KBM-03 代码级唯一真源

// MaxSlugLen 是 slug 长度上限的单一 home（=20，对齐 Milvus tier VarChar 列宽）。
// internal/retriever/milvus/config.go 的 DefaultTierMaxLen 将改为引用本常量（26-03）。
const MaxSlugLen = 20

type Entry struct {
    Slug             string    // ^[a-z][a-z0-9-]{0,19}$
    DisplayName      string    // UI 徽标文本（seed: 权威/历史/个人/草稿）
    Role             Role
    Description      string
    RetrievalEnabled bool      // false = 停用（不物理删除，KBM-02）
    CreatedAt        time.Time
    UpdatedAt        time.Time
}

type Registry interface {
    Lookup(slug string) (Entry, bool)  // 仅活跃条目；停用 = 不存在（fail-closed）
    List() []Entry                     // 全部条目含停用（admin 视图），按 slug 升序
    ActiveSlugs() []string             // RetrievalEnabled==true，升序
    SlugsByRole(r Role) []string       // 活跃且 role==r，升序
    RetrievalEligible() []string       // 活跃 且 role != draft，升序（检索过滤集合原料）
    Signature() string                 // sha256 内容签名（审计行携带）
    ResolveReads(reads []string) ([]string, error)  // KBM-04：<slug>/** | <slug> | role:<r>
}

// Store 是注册表的持久化 seam（26-02 由 internal/store/sqlite 实现；
// 本 plan 测试用内存 fake）。kb 包不 import internal/store。
type Store interface {
    LoadBases(ctx context.Context) ([]Entry, error)
    UpsertBase(ctx context.Context, e Entry) error
    SetRetrievalEnabled(ctx context.Context, slug string, enabled bool, updatedAt time.Time) error
}

// slug.go
func ValidateSlug(s string) error  // 正则 ^[a-z][a-z0-9-]{0,19}$ + 保留字拒绝

// seed.go
func LoadSeedFile(path string) ([]Entry, error)             // absent → (nil, nil)；malformed → error
func LoadSeedRegistry(path string) (Registry, error)        // LoadSeedFile + New（cmd 二进制便捷构造）

// registry.go
func New(entries []Entry) *Registry                          // 只读快照；Store 为 nil
func (r *Registry) Create(in CreateInput) (Entry, error)     // Store==nil → ErrReadOnly
func (r *Registry) Deactivate(slug string) (Entry, error)    // Store==nil → ErrReadOnly
type CreateInput struct { Slug, DisplayName string; Role Role; Description string }
var ErrReadOnly = errors.New("kb: registry opened read-only (no Store)")
</interfaces>
