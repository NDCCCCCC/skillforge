import type { DataReference, PointReference } from '@/types/wire-types';

interface DataReferencePopupProps {
  dataRef: DataReference;
  pointRef?: PointReference;
}

export function DataReferencePopup({ dataRef, pointRef }: DataReferencePopupProps) {
  const displayValue = pointRef
    ? Number(pointRef.value).toLocaleString()
    : Number(dataRef.asOfPeriod).toLocaleString(); // fallback for single-value
  const period = pointRef?.period ?? dataRef.asOfPeriod;
  const pointInfo = pointRef && dataRef.references
    ? `（点 ${pointRef.pointIndex + 1} / ${dataRef.references.length}）`
    : '';

  return (
    <div className="flex flex-col gap-space-1 p-space-2 min-w-0 max-w-[280px]" data-testid="data-ref-popup">
      <span className="text-[var(--color-text)] text-sm font-medium">
        {displayValue}{pointInfo}
      </span>
      <span className="text-[var(--color-text-muted)] text-xs">
        周期：{period}
      </span>
      <span className="text-[var(--color-text-muted)] text-xs">
        数据截至：{dataRef.asOfTs}
      </span>
      <span className="text-[var(--color-text-dim)] text-xs">
        来源：data:aggregated_only
      </span>
      <span className="text-[var(--color-text-muted)] text-xs font-mono text-[10px]">
        函数：{dataRef.func}
      </span>
    </div>
  );
}
```

import { Tooltip, TooltipContent, TooltipTrigger } from '@/components/ui/tooltip';
import type { DataReference } from '@/types/wire-types';
import { DataReferencePopup } from './DataReferencePopup';
import { Sparkline } from './Sparkline';

interface DataReferenceChipProps {
  dataRef: DataReference;
}

export function DataReferenceChip({ dataRef }: DataReferenceChipProps) {
  const refs = dataRef.references ?? [];
  const hasSparkline = refs.length > 1;

  if (hasSparkline) {
    return (
      <span className="kpi-sparkline-wrapper inline-flex flex-col gap-space-1" data-testid="kpi-sparkline-ref">
        <Sparkline
          points={refs.map((r, i) => ({
            index: i,
            period: r.period,
            value: Number(r.value),
          }))}
          onPointHover={(i) => {/* tooltip handled by parent */}}
        />
      </span>
    );
  }

  // Single-value: inline span with tooltip
  const ref = refs[0];
  const displayValue = ref ? Number(ref.value).toLocaleString() : '—';

  return (
    <Tooltip>
      <TooltipTrigger asChild>
        <span className="kpi-value-inline cursor-help underline decoration-dotted" data-testid="kpi-value">
          {displayValue}
        </span>
      </TooltipTrigger>
      <TooltipContent side="top" className="max-w-[280px] p-0 border border-[var(--color-border)] rounded-md bg-[var(--color-surface-2)]">
        <DataReferencePopup dataRef={dataRef} pointRef={ref} />
      </TooltipContent>
    </Tooltip>
  );
}
```
