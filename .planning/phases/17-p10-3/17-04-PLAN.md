```go
package kpi

import "context"

// ResolvePeriods returns the sorted intersection of reqPeriods and the periods
// registered for dataset in the datasets registry. Every period in the result
// exists in the registry (i.e. has been imported). If a requested period is
// missing, the caller returns ReasonDatasetNotAvailable.
func ResolvePeriods(ctx context.Context, meta DatasetMetaSource, dataset string, reqPeriods []string) ([]string, error) {
    if len(reqPeriods) == 0 {
        return nil, nil
    }
    available, err := meta.Periods(ctx, dataset)
    if err != nil {
        return nil, err
    }
    // Build set of available periods
    availSet := make(map[string]struct{}, len(available))
    for _, p := range available {
        availSet[p] = struct{}{}
    }
    // Intersect
    var result []string
    for _, p := range reqPeriods {
        if _, ok := availSet[p]; ok {
            result = append(result, p)
        } else {
            return nil, &Refusal{ReasonCode: ReasonDatasetNotAvailable, HumanExplanation: "数据集 "+dataset+" 在周期 "+p+" 无数据"}
        }
    }
    sortStrings(result)
    return result, nil
}

func sortStrings(ss []string) {
    // standard sort
    for i := 0; i < len(ss)-1; i++ {
        for j := i + 1; j < len(ss); j++ {
            if ss[i] > ss[j] {
                ss[i], ss[j] = ss[j], ss[i]
            }
        }
    }
}
```

```go
// buildPointReferences converts snap.Rows to per-point references for sparkline.
// Each row has a `period` column tagged at executor step ⑤. D17-B2.
func buildPointReferences(rows []map[string]any) []wire.PointReference {
    out := make([]wire.PointReference, 0, len(rows))
    for i, row := range rows {
        p := wire.PointReference{PointIndex: i}
        if period, ok := row["period"].(string); ok {
            p.Period = period
        }
        // Last non-period, non-row_count column = the value
        for k, v := range row {
            if k != "period" && k != "row_count" {
                p.Value = v
                break
            }
        }
        out = append(out, p)
    }
    return out
}
```
