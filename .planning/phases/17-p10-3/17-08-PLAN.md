```yaml
# Used for GEVL-09/10 eval + reid frozen subset.
# DO NOT use in production — this is a synthetic fixture.

version: "1"
dataset: testdata

kpis:
  - name: KPI_LOAN_BALANCE
    description: 贷款余额合计（所有未偿还本金）
    unit: 亿元
    decimal_places: 2
    params: {}
    sql: |
      SELECT SUM(balance) AS value FROM testdata WHERE period = :period

  - name: KPI_NPL_RATIO
    description: 不良贷款率（不良贷款余额/总贷款余额）
    unit: "%"
    decimal_places: 3
    params: {}
    sql: |
      SELECT CAST(SUM(CASE WHEN status = 'NPL' THEN balance ELSE 0 END) AS DOUBLE) /
             NULLIF(SUM(balance), 0) AS value
      FROM testdata WHERE period = :period

  - name: KPI_INTEREST_INCOME
    description: 利息收入合计
    unit: 亿元
    decimal_places: 2
    params: {}
    sql: |
      SELECT SUM(interest_income) AS value FROM testdata WHERE period = :period

  - name: KPI_CUSTOMER_COUNT
    description: 活跃客户数
    unit: 户
    decimal_places: 0
    params: {}
    sql: |
      SELECT COUNT(DISTINCT customer_id) AS value FROM testdata WHERE period = :period

  - name: KPI_LOAN_BALANCE_TREND
    description: 贷款余额多周期趋势（用于sparkline）
    unit: 亿元
    decimal_places: 2
    params:
      periods:
        type: array
        required: true
        items:
          type: string
          pattern: '^\d{4}-(0[1-9]|1[0-2])$'
          min_items: 2
          max_items: 12
    sql: |
      SELECT period, SUM(balance) AS value
      FROM testdata
      WHERE period = ANY(:periods)
      GROUP BY period
      ORDER BY period ASC
```

```go
// gen_testdata.go — generates data/staging/testdata.duckdb
package main

import (
    "database/sql"
    "fmt"
    "math/rand"
    _ "github.com/duckdb/duckdb-go"
)

const seed = 42

func main() {
    db, _ := sql.Open("duckdb", "data/staging/testdata.duckdb")
    defer db.Close()

    // Create table
    db.Exec(`CREATE TABLE testdata (
        period VARCHAR,
        customer_id VARCHAR,
        dimension VARCHAR,
        balance DOUBLE,
        status VARCHAR,
        interest_income DOUBLE
    )`)

    rand.Seed(seed)
    dimensions := []string{"银行A", "银行B", "银行C", "银行D"}
    periods := []string{
        "2025-08","2025-09","2025-10","2025-11","2025-12",
        "2026-01","2026-02","2026-03","2026-04","2026-05","2026-06","2026-07",
    }
    status := []string{"normal", "normal", "normal", "normal", "NPL"} // 20% NPL rate

    // 208 rows per (period, dimension) ≈ 10,000 total
    perBucket := 208
    customerID := 1

    for _, dim := range dimensions {
        for _, period := range periods {
            for i := 0; i < perBucket; i++ {
                bal := 100000 + rand.Float64()*9900000 // 10万 ~ 1亿
                stat := status[rand.Intn(len(status))]
                interest := bal * (0.03 + rand.Float64()*0.02) // 3-5% 利率
                cid := fmt.Sprintf("C%05d", customerID)
                customerID++
                db.Exec(`INSERT INTO testdata VALUES (?,?,?,?,?,?)`,
                    period, cid, dim, bal, stat, interest)
            }
        }
    }
    fmt.Printf("Generated %d rows\n", customerID-1)
}
```

Run this to generate the duckdb file. The fixture should be committed to the repo (per Phase 16 precedent of committing WSL test fixtures).

Alternative: if DuckDB CLI is available in WSL, use:
```sql
CREATE TABLE testdata AS SELECT ...;
```

The key invariant: every (period, dimension) bucket must have >= 50 rows so k_min=5 suppression never triggers on the expected KPI values used in the golden set.
