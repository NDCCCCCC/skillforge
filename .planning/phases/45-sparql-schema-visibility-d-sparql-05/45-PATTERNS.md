```go
func isSchemaTriple(t Triple) bool {
	// rdf:type for OWL vocabulary declarations.
	if t.P == rdfType {
		if t.O == ns+"Class" || t.O == ns+"ObjectProperty" ||
			t.O == ns+"DatatypeProperty" || t.O == ns+"Property" ||
			t.O == ns+"NamedIndividual" {
			return true
		}
		// rdf:type with a schema individual as object (skf:Skill, skf:Tier, etc.)
		if schemaClasses[t.O] {
			return true
		}
	}

	// OWL/RDFS ontology metadata predicates — always schema layer.
	// label/comment are intentionally EXCLUDED (D-SPARQL-03) ...
	if t.P == ns+"domain" || t.P == ns+"range" ||
		t.P == ns+"subClassOf" || t.P == ns+"subPropertyOf" {
		return true
	}

	return false
}
```

```go
func (s *Store) LoadFromTurtle(ttl string) (int, error) {
	g := graph.NewGraph()
	if err := turtle.Parse(g, strings.NewReader(ttl)); err != nil {
		return 0, fmt.Errorf("store: parse turtle: %w", err)
	}
	n := 0
	for tr := range g.Triples(nil, nil, nil) {
		s.Insert(Triple{
			S: subjectString(tr.Subject),
			P: tr.Predicate.Value(),
			O: objectString(tr.Object),
		})
		n++
	}
	return n, nil
}
```
