```go
// QueuedTickets returns tickets with dispatch_status='queued' created before olderThan.
func (db *DB) QueuedTickets(ctx context.Context, olderThan time.Time) ([]*TicketRow, error) {
    rows, err := db.db.QueryContext(ctx, `
        SELECT id, session_id, turn_id, description, chunk_ids, skill_id,
               ticket_number, dispatch_status, created_at
        FROM tickets
        WHERE dispatch_status = 'queued' AND created_at < ?
    `, olderThan)
    if err != nil {
        return nil, err
    }
    defer rows.Close()
    var out []*TicketRow
    for rows.Next() {
        var t TicketRow
        // scan into t fields
        out = append(out, &t)
    }
    return out, rows.Err()
}
```

**Part B — tickets.go 启动 sweep：**

在 `NewServer` 或 server 启动时调用 sweep：
1. 定义 sweep 阈值（如 `time.Now().Add(-5 * time.Minute)` — 5 分钟内的工单不重发，超过 5 分钟的重新外发）

**INSERT-only 纪律**：sweep 不更新 ticket 行 dispatch_status，只通过 dispatchForTicket 发起新的外发尝试并产生 `action_ticket_dispatch_started` 审计行。

**Part A — rate limiter 结构体（Server struct 上）：**

在 `Server` struct 上添加：
```go
import "golang.org/x/time/rate"

// server.go 中 Server struct 添加：
type Server struct {
    // ... 现有字段 ...
    feedbackLimiter *rate.Limiter  // per-session feedback limiter
    ticketLimiter   *rate.Limiter  // per-session ticket limiter (shared across sessions, actual limit is per-token)
    // 或使用 map[sessionID]*rate.Limiter
}
```

实际实现推荐用 `sync.Map[sessionID]*rate.Limiter` 每个 session 一个 limiter，限制例如每分钟 10 个请求：
```go
type perSessionLimiters struct {
    mu sync.Mutex
    limiters map[string]*rate.Limiter
}

func (m *perSessionLimiters) Allow(sessionID string) bool {
    m.mu.Lock()
    defer m.mu.Unlock()
    lim, ok := m.limiters[sessionID]
    if !ok {
        lim = rate.NewLimiter(rate.Every(time.Minute), 10) // 10 per minute
        m.limiters[sessionID] = lim
    }
    return lim.Allow()
}
```
